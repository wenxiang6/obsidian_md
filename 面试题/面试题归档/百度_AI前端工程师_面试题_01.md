```javascript
class StateMachineDebugger {
  constructor(stateMachine) {
    this.stateMachine = stateMachine;
    this.history = [];
    this.watchpoints = new Set();
    this.breakpoints = new Map();
  }
  
  // 1. 历史记录追踪
  trackHistory() {
    const originalTrigger = this.stateMachine.trigger.bind(this.stateMachine);
    
    this.stateMachine.trigger = async (event, data) => {
      const prevState = this.stateMachine.currentState;
      const timestamp = Date.now();
      
      console.log(`[${timestamp}] 状态转换: ${prevState} -> ${event}`);
      
      try {
        const result = await originalTrigger(event, data);
        
        this.history.push({
          timestamp,
          from: prevState,
          to: this.stateMachine.currentState,
          event,
          data,
          duration: Date.now() - timestamp
        });
        
        // 检测异常模式
        this.detectAnomalies();
        
        return result;
      } catch (error) {
        console.error(`状态转换失败: ${error.message}`);
        throw error;
      }
    };
  }
  
  // 2. 异常模式检测
  detectAnomalies() {
    // 检测死循环
    if (this.isInfiniteLoop()) {
      console.error('检测到死循环！');
      this.handleInfiniteLoop();
    }
    
    // 检测卡死状态
    if (this.isStuck()) {
      console.warn('检测到状态卡死');
      this.handleStuckState();
    }
    
    // 检测异常频繁转换
    if (this.isRapidTransitions()) {
      console.warn('检测到异常频繁的状态转换');
      this.handleRapidTransitions();
    }
  }
  
  // 3. 死循环检测
  isInfiniteLoop() {
    const recentHistory = this.history.slice(-10);
    if (recentHistory.length < 5) return false;
    
    // 检测重复模式
    const pattern = recentHistory.map(h => `${h.from}->${h.to}`).join(',');
    const patterns = this.history.slice(-20).map(h => `${h.from}->${h.to}`).join(',');
    
    // 检查是否有重复的转换序列
    const sequenceLength = 3;
    for (let i = 0; i < recentHistory.length - sequenceLength; i++) {
      const sequence = recentHistory.slice(i, i + sequenceLength)
        .map(h => h.event).join(',');
      
      const nextSequence = recentHistory.slice(i + 1, i + 1 + sequenceLength)
        .map(h => h.event).join(',');
      
      if (sequence === nextSequence) {
        return true;
      }
    }
    
    return false;
  }
  
  // 4. 卡死状态检测
  isStuck() {
    if (this.history.length === 0) return false;
    
    const lastTransition = this.history[this.history.length - 1];
    const timeSinceLastTransition = Date.now() - lastTransition.timestamp;
    
    // 如果超过30秒没有状态转换，认为卡死
    return timeSinceLastTransition > 30000;
  }
  
  // 5. 异常频繁转换检测
  isRapidTransitions() {
    const recentHistory = this.history.slice(-5);
    if (recentHistory.length < 5) return false;
    
    const totalTime = recentHistory[recentHistory.length - 1].timestamp - recentHistory[0].timestamp;
    const transitionsPerSecond = recentHistory.length / (totalTime / 1000);
    
    // 如果每秒超过10次转换，认为异常频繁
    return transitionsPerSecond > 10;
  }

    console.error('执行死循环恢复策略');
    
    // 策略1：强制重置状态
    this.stateMachine.currentState = 'idle';
    this.stateMachine.context = {};
    
    // 策略2：通知用户
    this.notifyUser('检测到死循环，已自动重置');

      history: this.history.slice(-20),
      timestamp: Date.now()
    });
  }
  
  // 7. 处理卡死状态
  handleStuckState() {
    console.warn('执行卡死状态恢复策略');
    
    // 策略1：发送心跳检查
    this.sendHeartbeat();

    setTimeout(() => {
      if (this.isStuck()) {
        this.stateMachine.currentState = 'idle';
        this.notifyUser('状态卡死，已自动恢复');
      }
    }, 10000);
  }
  
  // 8. 处理异常频繁转换
  handleRapidTransitions() {
    console.warn('执行限流策略');
    
    // 添加延迟
    const originalTrigger = this.stateMachine.trigger;
    let lastTriggerTime = 0;
    
    this.stateMachine.trigger = async (event, data) => {
      const now = Date.now();
      if (now - lastTriggerTime < 100) {
        console.warn('触发限流，延迟执行');
        await new Promise(resolve => setTimeout(resolve, 100));
      }
      lastTriggerTime = Date.now();
      
      return originalTrigger(event, data);
    };
  }
  
  // 9. 监视点
  addWatchpoint(condition, callback) {
    this.watchpoints.add({ condition, callback });
  }
  
  checkWatchpoints() {
    for (const watchpoint of this.watchpoints) {
      if (watchpoint.condition(this.stateMachine)) {
        watchpoint.callback(this.stateMachine);
      }
    }
  }
  
  // 10. 断点
  addBreakpoint(state, callback) {
    if (!this.breakpoints.has(state)) {
      this.breakpoints.set(state, []);
    }
    this.breakpoints.get(state).push(callback);
  }
  
  checkBreakpoints(state) {
    const callbacks = this.breakpoints.get(state) || [];
    callbacks.forEach(callback => callback(this.stateMachine));
  }
}

// 熔断机制
class CircuitBreaker {
  constructor(options = {}) {
    this.state = 'closed'; // closed, open, half-open
    this.failureCount = 0;
    this.successCount = 0;
    this.lastFailureTime = null;
    
    this.options = {
      failureThreshold: 5, // 失败阈值
      successThreshold: 3, // 成功阈值
      timeout: 60000, // 超时时间
      ...options
    };
  }
  
  async execute(fn) {
    if (this.state === 'open') {
      if (this.shouldAttemptReset()) {
        this.state = 'half-open';
      } else {
        throw new Error('熔断器打开，拒绝请求');
      }
    }
    
    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }
  
  onSuccess() {
    if (this.state === 'half-open') {
      this.successCount++;
      if (this.successCount >= this.options.successThreshold) {
        this.reset();
      }
    } else {
      this.failureCount = 0;
    }
  }
  
  onFailure() {
    this.failureCount++;
    this.lastFailureTime = Date.now();
    
    if (this.failureCount >= this.options.failureThreshold) {
      this.trip();
    }
  }
  
  trip() {
    this.state = 'open';
    console.warn('熔断器打开');
  }
  
  reset() {
    this.state = 'closed';
    this.failureCount = 0;
    this.successCount = 0;
    console.log('熔断器关闭');
  }
  
  shouldAttemptReset() {
    return Date.now() - this.lastFailureTime >= this.options.timeout;
  }
  
  getState() {
    return {
      state: this.state,
      failureCount: this.failureCount,
      successCount: this.successCount,
      lastFailureTime: this.lastFailureTime
    };
  }
}
```

## 2. FFmpeg剪辑CPU/内存突增，Agent动态限流、任务分片降级方案？

**FFmpeg资源管理：**

```javascript
class FFmpegResourceManager {
  constructor() {
    this.tasks = new Map();
    this.resourceMonitor = new ResourceMonitor();
    this.rateLimiter = new RateLimiter();
    this.taskQueue = new PriorityQueue();
  }
  
  // 1. 资源监控
  async monitorResources() {
    const usage = await this.resourceMonitor.getUsage();
    
    console.log('当前资源使用:', {
      cpu: `${usage.cpu}%`,
      memory: `${usage.memory}%`,
      disk: `${usage.disk}%`,
      activeTasks: this.tasks.size
    });
    
    // 检查是否需要限流
    if (this.shouldThrottle(usage)) {
      await this.throttleTasks();
    }
    
    // 检查是否需要降级
    if (this.shouldDegrade(usage)) {
      await this.degradeTasks();
    }
    
    return usage;
  }
  
  // 2. 动态限流
  shouldThrottle(usage) {
    return usage.cpu > 80 || usage.memory > 85 || this.tasks.size > 5;
  }
  
  async throttleTasks() {
    console.log('执行限流策略');
    
    // 降低并发数
    this.rateLimiter.reduceConcurrency();
    
    // 延迟低优先级任务
    const lowPriorityTasks = Array.from(this.tasks.values())
      .filter(task => task.priority === 'low');
    
    for (const task of lowPriorityTasks) {
      await this.delayTask(task, 5000);
    }
  }
  
  // 3. 任务分片
  async splitTask(task) {
    const { input, output, duration } = task;
    
    // 根据视频时长分片
    const chunkDuration = 60; // 60秒一片
    const chunks = [];
    
    for (let start = 0; start < duration; start += chunkDuration) {
      const end = Math.min(start + chunkDuration, duration);
      chunks.push({
        id: `${task.id}-${start}`,
        input,
        output: `${output}-${start}`,
        startTime: start,
        endTime: end,
        priority: task.priority,
        parentTask: task.id
      });
    }
    
    return chunks;
  }
  
  // 4. 降级策略
  shouldDegrade(usage) {
    return usage.cpu > 90 || usage.memory > 95;
  }
  
  async degradeTasks() {
    console.log('执行降级策略');
    
    // 降低视频质量
    const activeTasks = Array.from(this.tasks.values())
      .filter(task => task.status === 'processing');
    
    for (const task of activeTasks) {
      await this.degradeTaskQuality(task);
    }
  }
  
  async degradeTaskQuality(task) {
    // 降低分辨率
    if (task.resolution === '1080p') {
      task.resolution = '720p';
      task.quality = 'medium';
      console.log(`任务 ${task.id} 降级到 720p`);
    } else if (task.resolution === '720p') {
      task.resolution = '480p';
      task.quality = 'low';
      console.log(`任务 ${task.id} 降级到 480p`);
    }
  }
  
  // 5. 任务优先级队列
  async addTask(taskConfig) {
    const task = {
      id: `task-${Date.now()}`,
      ...taskConfig,
      status: 'queued',
      createdAt: Date.now()
    };
    
    // 根据优先级插入队列
    this.taskQueue.enqueue(task, this.getPriorityScore(task));
    
    // 尝试执行任务
    await this.processQueue();
    
    return task;
  }
  
  getPriorityScore(task) {
    const priorities = { high: 3, medium: 2, low: 1 };
    return priorities[task.priority] || 1;
  }
  
  async processQueue() {
    while (!this.taskQueue.isEmpty() && this.canProcessMore()) {
      const task = this.taskQueue.dequeue();
      await this.executeTask(task);
    }
  }
  
  canProcessMore() {
    const usage = this.resourceMonitor.getLastUsage();
    return this.tasks.size < 5 && usage.cpu < 70 && usage.memory < 80;
  }
  
  // 6. 任务执行
  async executeTask(task) {
    this.tasks.set(task.id, task);
    task.status = 'processing';
    task.startedAt = Date.now();
    
    try {
      // 构建FFmpeg命令
      const command = this.buildFFmpegCommand(task);
      
      // 执行FFmpeg
      const result = await this.executeFFmpeg(command);
      
      task.status = 'completed';
      task.completedAt = Date.now();
      task.duration = task.completedAt - task.startedAt;
      
      console.log(`任务 ${task.id} 完成，耗时 ${task.duration}ms`);
      
    } catch (error) {
      task.status = 'failed';
      task.error = error.message;
      console.error(`任务 ${task.id} 失败:`, error);
      
      // 重试逻辑
      if (task.retryCount < 3) {
        task.retryCount = (task.retryCount || 0) + 1;
        console.log(`重试任务 ${task.id}，第 ${task.retryCount} 次`);
        await this.delayTask(task, 2000 * task.retryCount);
      }
    } finally {
      this.tasks.delete(task.id);
      await this.processQueue();
    }
  }
  
  buildFFmpegCommand(task) {
    const { input, output, resolution, quality, startTime, endTime } = task;
    
    const qualitySettings = {
      high: { crf: 18, preset: 'slow' },
      medium: { crf: 23, preset: 'medium' },
      low: { crf: 28, preset: 'fast' }
    };
    
    const settings = qualitySettings[quality] || qualitySettings.medium;
    
    let command = `ffmpeg -i ${input}`;
    
    if (startTime !== undefined) {
      command += ` -ss ${startTime}`;
    }
    
    if (endTime !== undefined) {
      command += ` -to ${endTime}`;
    }
    
    command += ` -vf scale=${this.getResolutionScale(resolution)}`;
    command += ` -c:v libx264 -crf ${settings.crf} -preset ${settings.preset}`;
    command += ` -c:a aac -b:a 128k`;
    command += ` ${output}`;
    
    return command;
  }
  
  getResolutionScale(resolution) {
    const scales = {
      '1080p': '1920:1080',
      '720p': '1280:720',
      '480p': '854:480'
    };
    return scales[resolution] || scales['720p'];
  }
  
  async executeFFmpeg(command) {
    return new Promise((resolve, reject) => {
      const { exec } = require('child_process');
      
      exec(command, (error, stdout, stderr) => {
        if (error) {
          reject(new Error(`FFmpeg执行失败: ${error.message}`));
        } else {
          resolve({ stdout, stderr });
        }
      });
    });
  }
  
  async delayTask(task, delay) {
    return new Promise(resolve => {
      setTimeout(async () => {
        await this.processQueue();
        resolve();
      }, delay);
    });
  }
}

class ResourceMonitor {
  constructor() {
    this.lastUsage = null;
    this.monitoring = false;
  }
  
  async getUsage() {
    const os = require('os');
    const fs = require('fs');
    
    // CPU使用率
    const cpus = os.cpus();
    const cpuUsage = cpus.reduce((acc, cpu) => {
      const total = Object.values(cpu.times).reduce((a, b) => a + b);
      const idle = cpu.times.idle;
      return acc + ((total - idle) / total) * 100;
    }, 0) / cpus.length;
    
    // 内存使用率
    const totalMem = os.totalmem();
    const freeMem = os.freemem();
    const memUsage = ((totalMem - freeMem) / totalMem) * 100;
    
    // 磁盘使用率（简化实现）
    const diskUsage = 50; // 需要实际实现
    
    this.lastUsage = {
      cpu: Math.round(cpuUsage),
      memory: Math.round(memUsage),
      disk: diskUsage,
      timestamp: Date.now()
    };
    
    return this.lastUsage;
  }
  
  getLastUsage() {
    return this.lastUsage || { cpu: 0, memory: 0, disk: 0 };
  }
  
  startMonitoring(interval = 5000) {
    this.monitoring = true;
    
    const monitor = async () => {
      if (!this.monitoring) return;
      
      await this.getUsage();
      setTimeout(monitor, interval);
    };
    
    monitor();
  }
  
  stopMonitoring() {
    this.monitoring = false;
  }
}

class RateLimiter {
  constructor() {
    this.concurrency = 5;
    this.minConcurrency = 1;
    this.maxConcurrency = 10;
  }
  
  reduceConcurrency() {
    this.concurrency = Math.max(this.minConcurrency, this.concurrency - 1);
    console.log(`并发数降低到: ${this.concurrency}`);
  }
  
  increaseConcurrency() {
    this.concurrency = Math.min(this.maxConcurrency, this.concurrency + 1);
    console.log(`并发数增加到: ${this.concurrency}`);
  }
  
  getConcurrency() {
    return this.concurrency;
  }
}

async function processBatch(videos) {

    input: video.input,
    output: video.output,

  }));

        await resourceManager.addTask(chunk);

    }
  }
```

## 3. 切换国产大模型，提示词、工具调用逻辑的改动点？

```javascript
class ModelAdapter {
  constructor() {
    this.models = new Map();
    this.promptTemplates = new Map();
    this.toolDefinitions = new Map();
  }
  
  // 1. 注册模型
  registerModel(modelConfig) {
    const { id, name, provider, capabilities, limitations } = modelConfig;
    
    this.models.set(id, {
      id,
      name,

  // 2. 提示词适配
  adaptPrompt(prompt, modelId) {

    const template = this.promptTemplates.get(modelId) || this.promptTemplates.get('default');
    
    // 根据模型特点调整提示词
    const adaptedPrompt = this.applyModelSpecificRules(prompt, model);

  }
  
  applyModelSpecificRules(prompt, model) {
    let adapted = prompt;
    
    // 国产模型通常需要更明确的指令
    if (model.provider === 'baidu' || model.provider === 'alibaba') {

      if (!adapted.includes('你是')) {

      if (adapted.includes('JSON') && !adapted.includes('请以JSON格式')) {

      }
    }
    
    // 通义千问需要特殊处理
    if (model.provider === 'alibaba') {
      // 通义千问对系统消息的处理方式不同
      adapted = adapted.replace(/system:/g, '系统指令:');
    }
    
    // 文心一言需要特殊处理
    if (model.provider === 'baidu') {
      // 文心一言对工具调用的格式要求不同
      if (adapted.includes('function_calling')) {
        adapted = this.adaptForWenxin(adapted);
      }
    }
    
    return adapted;
  }
  
  // 3. 工具调用适配
  adaptToolDefinition(tool, modelId) {
    const model = this.models.get(modelId);
    const adaptedTool = { ...tool };
    
    // 根据模型调整工具定义
    switch (model.provider) {
      case 'openai':
        // OpenAI格式
        return this.adaptForOpenAI(adaptedTool);
      
      case 'baidu':
        // 文心一言格式
        return this.adaptForWenxin(adaptedTool);
      
      case 'alibaba':

        return this.adaptForQwen(adaptedTool);
      
      case 'zhipu':
        // 智谱AI格式
        return this.adaptForZhipu(adaptedTool);
      
      default:
        return adaptedTool;
    }
  }
  
  adaptForOpenAI(tool) {
    return {
      type: 'function',
      function: {
        name: tool.name,
        description: tool.description,
        parameters: tool.parameters
      }
    };
  }
  
  adaptForWenxin(tool) {

    return {
      name: tool.name,
      description: tool.description,
      parameters: tool.parameters,
      // 文心一言需要额外的字段
      response_format: 'json'
    };
  }
  
  adaptForQwen(tool) {
    // 通义千问的工具定义格式
    return {
      type: 'function',
      function: {
        name: tool.name,
        description: tool.description,
        parameters: {
          type: 'object',
          properties: tool.parameters.properties,
          required: tool.parameters.required || []
        }
      }
    };
  }
  
  adaptForZhipu(tool) {
    // 智谱AI的工具定义格式
    return {
      name: tool.name,
      description: tool.description,
      parameters: tool.parameters,
      // 智谱AI需要的额外配置
      return_direct: false
    };
  }
  
  // 4. 响应解析适配
  parseResponse(response, modelId) {
    const model = this.models.get(modelId);
    
    switch (model.provider) {
      case 'openai':
        return this.parseOpenAIResponse(response);
      
      case 'baidu':
        return this.parseWenxinResponse(response);
      
      case 'alibaba':
        return this.parseQwenResponse(response);
      
      case 'zhipu':
        return this.parseZhipuResponse(response);
      
      default:
        return response;
    }
  }

    return {
      content: response.choices[0].message.content,
      toolCalls: response.choices[0].message.tool_calls,
      usage: response.usage
    };
  }

    return {
      content: response.result,
      toolCalls: response.tool_calls || [],
      usage: {
        prompt_tokens: response.usage.prompt_tokens,
        completion_tokens: response.usage.completion_tokens,
        total_tokens: response.usage.total_tokens
      }
    };
  }
  
  parseQwenResponse(response) {

    return {
      content: response.output.text,
      toolCalls: response.output.tool_calls || [],
      usage: response.usage
    };
  }
  
  parseZhipuResponse(response) {
    // 智谱AI的响应格式
    return {
      content: response.choices[0].message.content,
      toolCalls: response.choices[0].message.tool_calls || [],
      usage: response.usage
    };
  }
  
  // 5. 统一调用接口
  async chat(messages, options = {}) {
    const { modelId, tools, temperature, maxTokens } = options;
    const model = this.models.get(modelId);
    
    // 适配消息格式

    // 调用模型API

      messages: adaptedMessages,
      tools: adaptedTools,
      temperature,
      max_tokens: maxTokens
    });
    
    // 解析响应
    return this.parseResponse(response, modelId);
  }
  
  adaptMessages(messages, modelId) {
    const model = this.models.get(modelId);
    
    return messages.map(message => {
      // 根据模型调整消息格式

      }
      
      return message;
    });
  }

      method: 'POST',
      headers: {

  }
```

## 4. 海量积压场景，RabbitMQ死信队列、延迟队列如何保障消费稳定性？

**答案：**
**RabbitMQ高可用架构：**

```javascript
class RabbitMQManager {
  constructor(config) {
    this.config = config;
    this.connection = null;
    this.channel = null;
    this.consumers = new Map();
    this.deadLetterQueue = 'dead_letter_queue';
    this.retryQueue = 'retry_queue';
  }
  
  // 1. 连接管理
  async connect() {
    const amqp = require('amqplib');
    
    try {
      this.connection = await amqp.connect(this.config.url);
      this.channel = await this.connection.createChannel();
      
      // 设置预取数量，控制消费速度
      await this.channel.prefetch(this.config.prefetch || 10);
      
      // 监听连接事件
      this.connection.on('error', this.handleConnectionError.bind(this));
      this.connection.on('close', this.handleConnectionClose.bind(this));
      
      console.log('RabbitMQ连接成功');
      
    } catch (error) {
      console.error('RabbitMQ连接失败:', error);
      throw error;
    }
  }
  
  // 2. 队列配置
  async setupQueues() {
    // 死信队列配置
    await this.channel.assertExchange('dead_letter_exchange', 'direct', { durable: true });
    await this.channel.assertQueue(this.deadLetterQueue, {
      durable: true,
      arguments: {
        'x-message-ttl': 86400000, // 24小时
        'x-max-length': 1000000 // 最大消息数
      }
    });
    await this.channel.bindQueue(this.deadLetterQueue, 'dead_letter_exchange', 'dead');
    
    // 重试队列配置
    await this.channel.assertExchange('retry_exchange', 'direct', { durable: true });
    await this.channel.assertQueue(this.retryQueue, {
      durable: true,
      arguments: {
        'x-message-ttl': 60000, // 1分钟延迟
        'x-dead-letter-exchange': 'main_exchange', // 死信后转发到主交换机
        'x-dead-letter-routing-key': 'main'
      }
    });
    await this.channel.bindQueue(this.retryQueue, 'retry_exchange', 'retry');
    
    // 主队列配置
    await this.channel.assertExchange('main_exchange', 'direct', { durable: true });
    await this.channel.assertQueue('main_queue', {
      durable: true,
      arguments: {
        'x-dead-letter-exchange': 'dead_letter_exchange', // 死信转发
        'x-dead-letter-routing-key': 'dead',
        'x-max-priority': 10 // 优先级队列
      }
    });
    await this.channel.bindQueue('main_queue', 'main_exchange', 'main');
    
    console.log('队列配置完成');
  }
  
  // 3. 消息发布
  async publishMessage(message, options = {}) {
    const { priority = 5, delay = 0, retryCount = 0 } = options;
    
    const messageBuffer = Buffer.from(JSON.stringify(message));
    
    const publishOptions = {
      persistent: true,
      priority,
      timestamp: Date.now(),
      headers: {
        'x-retry-count': retryCount,
        'x-original-queue': 'main_queue'
      }
    };
    
    if (delay > 0) {
      // 延迟消息，发送到重试队列
      publishOptions.expiration = delay;
      await this.channel.publish('retry_exchange', 'retry', messageBuffer, publishOptions);
    } else {
      // 正常消息，发送到主队列
      await this.channel.publish('main_exchange', 'main', messageBuffer, publishOptions);
    }
    
    console.log(`消息已发布: ${message.id}`);
  }
  
  // 4. 消费者管理
  async consumeMessages(queue, handler, options = {}) {
    const { concurrency = 1, retryLimit = 3 } = options;
    
    for (let i = 0; i < concurrency; i++) {
      const consumerTag = `consumer-${queue}-${i}`;
      
      const wrappedHandler = async (msg) => {
        if (!msg) return;
        
        const message = JSON.parse(msg.content.toString());
        const retryCount = msg.properties.headers['x-retry-count'] || 0;
        
        try {
          // 处理消息
          await handler(message);
          
          // 确认消息
          this.channel.ack(msg);
          
          console.log(`消息处理成功: ${message.id}`);
          
        } catch (error) {
          console.error(`消息处理失败: ${message.id}`, error);
          
          if (retryCount < retryLimit) {
            // 重试，发送到重试队列
            await this.publishMessage(message, {
              ...options,
              delay: this.calculateRetryDelay(retryCount),
              retryCount: retryCount + 1
            });
            
            // 确认原消息
            this.channel.ack(msg);
            
          } else {
            // 超过重试次数，发送到死信队列
            await this.sendToDeadLetter(message, error);
            
            // 确认原消息
            this.channel.ack(msg);
          }
        }
      };
      
      await this.channel.consume(queue, wrappedHandler, {
        consumerTag,
        noAck: false
      });
      
      this.consumers.set(consumerTag, {
        queue,
        handler: wrappedHandler,
        startTime: Date.now()
      });
      
      console.log(`消费者 ${consumerTag} 已启动`);
    }
  }
  
  // 5. 重试延迟计算
  calculateRetryDelay(retryCount) {
    // 指数退避算法
    const baseDelay = 1000; // 1秒
    const maxDelay = 300000; // 5分钟
    
    const delay = Math.min(baseDelay * Math.pow(2, retryCount), maxDelay);
    
    // 添加随机抖动，避免惊群效应
    const jitter = delay * 0.1 * Math.random();
    
    return delay + jitter;
  }
  
  // 6. 死信队列处理
  async sendToDeadLetter(message, error) {
    const deadLetterMessage = {
      originalMessage: message,
      error: {
        message: error.message,
        stack: error.stack,
        timestamp: Date.now()
      },
      deadLetterReason: 'max_retries_exceeded'
    };
    
    await this.channel.publish(
      'dead_letter_exchange',
      'dead',
      Buffer.from(JSON.stringify(deadLetterMessage)),
      { persistent: true }
    );
    
    console.log(`消息已发送到死信队列: ${message.id}`);
  }
  
  // 7. 死信队列消费（人工处理）
  async consumeDeadLetter(handler) {
    await this.consumeMessages(this.deadLetterQueue, async (message) => {
      console.log('处理死信消息:', message.originalMessage.id);
      
      // 人工处理或记录日志
      await handler(message);
    });
  }
  
  // 8. 监控和告警
  async getQueueStats() {
    const mainQueue = await this.channel.checkQueue('main_queue');
    const retryQueue = await this.channel.checkQueue(this.retryQueue);
    const deadLetterQueue = await this.channel.checkQueue(this.deadLetterQueue);
    
    const stats = {
      mainQueue: {
        messages: mainQueue.messageCount,
        consumers: mainQueue.consumerCount
      },
      retryQueue: {
        messages: retryQueue.messageCount,
        consumers: retryQueue.consumerCount
      },
      deadLetterQueue: {
        messages: deadLetterQueue.messageCount,
        consumers: deadLetterQueue.consumerCount
      },
      timestamp: Date.now()
    };
    
    // 检查告警条件
    this.checkAlerts(stats);
    
    return stats;
  }
  
  checkAlerts(stats) {
    // 主队列积压告警
    if (stats.mainQueue.messages > 10000) {
      console.warn(`主队列积压告警: ${stats.mainQueue.messages} 条消息`);
      this.sendAlert('queue_backlog', stats);
    }
    
    // 死信队列告警
    if (stats.deadLetterQueue.messages > 100) {
      console.warn(`死信队列告警: ${stats.deadLetterQueue.messages} 条消息`);
      this.sendAlert('dead_letter_backlog', stats);
    }
    
    // 消费者数量告警
    if (stats.mainQueue.consumers === 0) {
      console.warn('主队列无消费者');
      this.sendAlert('no_consumers', stats);
    }
  }
  
  async sendAlert(type, data) {
    // 发送告警通知
    console.log(`发送告警: ${type}`, data);
  }
  
  // 9. 连接错误处理
  handleConnectionError(error) {
    console.error('RabbitMQ连接错误:', error);
    this.reconnect();
  }
  
  handleConnectionClose() {
    console.warn('RabbitMQ连接关闭');
    this.reconnect();
  }
  
  async reconnect() {
    console.log('尝试重新连接RabbitMQ...');
    
    let retries = 0;
    const maxRetries = 5;
    
    while (retries < maxRetries) {
      try {
        await this.connect();
        await this.setupQueues();
        console.log('RabbitMQ重连成功');
        return;
      } catch (error) {
        retries++;
        console.error(`重连失败 (${retries}/${maxRetries}):`, error);
        
        // 指数退避
        const delay = Math.min(1000 * Math.pow(2, retries), 30000);
        await new Promise(resolve => setTimeout(resolve, delay));
      }
    }
    
    console.error('RabbitMQ重连失败，达到最大重试次数');
  }
  
  // 10. 优雅关闭
  async close() {
    console.log('关闭RabbitMQ连接...');
    
    // 停止所有消费者
    for (const [tag, consumer] of this.consumers) {
      await this.channel.cancel(tag);
      console.log(`消费者 ${tag} 已停止`);
    }
    
    // 关闭通道和连接
    await this.channel.close();
    await this.connection.close();
    
    console.log('RabbitMQ连接已关闭');
  }
}
```

## 6. Context模式重构埋点在Vue中有何局限？

**答案：**
**Vue中Context模式的局限性：**

```javascript
// Vue 2/3 中的Provide/Inject模式
import { defineComponent, provide, inject, ref, onMounted } from 'vue';

// 1. 基本实现
const TrackingSymbol = Symbol('tracking');

function createTrackingProvider() {
  return defineComponent({
    setup() {
      const trackEvent = (eventName, data) => {
        console.log('Track:', eventName, data);
      };
      
      provide(TrackingSymbol, { trackEvent });
      
      return {};
    },

  });
}

function useTracking() {
  const tracking = inject(TrackingSymbol);
  
  if (!tracking) {
    throw new Error('useTracking must be used within a TrackingProvider');
  }
  
  return tracking;
}

// 2. 局限性分析
const limitations = {
  // 问题1：响应式丢失
  reactivityLoss: {
    description: 'provide/inject不是响应式的',
    example: `
      // 父组件
      const count = ref(0);
      provide('count', count); // ✅ 响应式
      provide('count', count.value); // ❌ 非响应式
      
      // 子组件
      const count = inject('count');
      // 如果provide的是值而不是ref，则不是响应式的
```

```javascript
  constructor() {
    this.phase = 'old'; // 'old', 'parallel', 'new'
    this.oldSystem = null;
    this.newSystem = null;
    this.featureFlags = new Map();
    this.metrics = {
      oldSystemCalls: 0,
      newSystemCalls: 0,
      errors: { old: 0, new: 0 },
      performance: { old: [], new: [] }
    };
  }
  
  // 1. 特性标志控制
  enableNewTracking(feature, percentage = 0) {
    this.featureFlags.set(feature, {
      enabled: percentage > 0,
      percentage,
      rolloutDate: Date.now()
    });
  }
  
  isFeatureEnabled(feature, userId) {
    const flag = this.featureFlags.get(feature);
    
    if (!flag || !flag.enabled) {
      return false;
    }
    
    // 基于用户ID的稳定分桶
    const bucket = this.getBucket(userId);
    return bucket < flag.percentage;
  }
  
  getBucket(userId) {
    // 简单的哈希分桶
    let hash = 0;
    for (let i = 0; i < userId.length; i++) {
      hash = ((hash << 5) - hash) + userId.charCodeAt(i);
      hash = hash & hash; // 转换为32位整数
    }
    return Math.abs(hash) % 100;
  }

  async trackEvent(eventName, data, context = {}) {
    const { userId, useNewSystem = false } = context;
    
    // 检查是否使用新系统
    const shouldUseNew = useNewSystem || 
      this.isFeatureEnabled(eventName, userId);
    
    // 并行执行两个系统
    const promises = [];
    
    // 始终执行旧系统（保证兼容性）
    promises.push(this.executeOldSystem(eventName, data));
    
    // 条件执行新系统
    if (shouldUseNew) {
      promises.push(this.executeNewSystem(eventName, data));
    }
    
    // 等待所有执行完成
    const results = await Promise.allSettled(promises);
    
    // 记录指标
    this.recordMetrics(eventName, results, shouldUseNew);
    
    // 返回旧系统的结果（保持兼容性）
    return results[0].status === 'fulfilled' ? results[0].value : null;
  }
  
  async executeOldSystem(eventName, data) {
    const startTime = performance.now();
    
    try {
      const result = await this.oldSystem.track(eventName, data);
      this.metrics.oldSystemCalls++;
      this.metrics.performance.old.push(performance.now() - startTime);
      return result;
    } catch (error) {
      this.metrics.errors.old++;
      console.error('旧系统埋点失败:', error);
      throw error;
    }
  }
  
  async executeNewSystem(eventName, data) {
    const startTime = performance.now();
    
    try {
      const result = await this.newSystem.track(eventName, data);
      this.metrics.newSystemCalls++;
      this.metrics.performance.new.push(performance.now() - startTime);
      return result;
    } catch (error) {
      this.metrics.errors.new++;
      console.error('新系统埋点失败:', error);
      // 新系统失败不影响主流程
      return null;
    }
  }
  
  // 3. 数据一致性验证
  async validateConsistency(eventName, data) {
    // 同时执行两个系统
    const [oldResult, newResult] = await Promise.all([
      this.executeOldSystem(eventName, data),
      this.executeNewSystem(eventName, data)
    ]);
    
    // 比较结果
    const isConsistent = this.compareResults(oldResult, newResult);
    
    if (!isConsistent) {
      console.warn(`数据不一致: ${eventName}`, {
        old: oldResult,
        new: newResult
      });
      
      // 记录不一致事件
      this.recordInconsistency(eventName, data, oldResult, newResult);
    }
    
    return isConsistent;
  }
  
  compareResults(oldResult, newResult) {
    // 简化比较逻辑
    if (!oldResult && !newResult) return true;
    if (!oldResult || !newResult) return false;
    
    // 比较关键字段
    const keysToCompare = ['event', 'timestamp', 'userId'];
    
    return keysToCompare.every(key => 
      JSON.stringify(oldResult[key]) === JSON.stringify(newResult[key])
    );
  }
  
  recordInconsistency(eventName, data, oldResult, newResult) {

      data,
      oldResult,
      newResult,
      timestamp: Date.now()
    });
  }
  
  // 4. 监控和告警

    const avgOld = this.metrics.performance.old.length > 0 
      ? this.metrics.performance.old.reduce((a, b) => a + b) / this.metrics.performance.old.length 
      : 0;
    
    const avgNew = this.metrics.performance.new.length > 0 
      ? this.metrics.performance.new.reduce((a, b) => a + b) / this.metrics.performance.new.length 
      : 0;
    
    return {
      totalCalls: this.metrics.oldSystemCalls + this.metrics.newSystemCalls,
      oldSystem: {
        calls: this.metrics.oldSystemCalls,
        errors: this.metrics.errors.old,
        avgResponseTime: avgOld
      },
      newSystem: {
        calls: this.metrics.newSystemCalls,
        errors: this.metrics.errors.new,
        avgResponseTime: avgNew
      },
      errorRate: {
        old: this.metrics.oldSystemCalls > 0 
          ? (this.metrics.errors.old / this.metrics.oldSystemCalls * 100).toFixed(2) + '%'
          : '0%',
        new: this.metrics.newSystemCalls > 0 
          ? (this.metrics.errors.new / this.metrics.newSystemCalls * 100).toFixed(2) + '%'
          : '0%'
      }
    };
  }

    const metrics = this.getMetrics();
    
    // 旧系统错误率告警
    if (parseFloat(metrics.errorRate.old) > 5) {
      this.sendAlert('old_system_high_error_rate', metrics);
    }
    
    // 新系统错误率告警
    if (parseFloat(metrics.errorRate.new) > 10) {

    }
    
    // 性能差异告警
    if (metrics.newSystem.avgResponseTime > metrics.oldSystem.avgResponseTime * 2) {
      this.sendAlert('new_system_slow_performance', metrics);
    }
  }

    console.warn(`告警: ${type}`, data);
    // 实际实现：发送到监控系统
  }
  
  // 5. 渐进式迁移
  async migratePhase(phase) {
    this.phase = phase;
    
    switch (phase) {
      case 'old':
        // 仅使用旧系统
        this.enableNewTracking('*', 0);
        break;
        
      case 'parallel':
        // 并行执行，新系统仅记录
        this.enableNewTracking('*', 100);
        break;
        
      case 'new':
        // 仅使用新系统
        this.enableNewTracking('*', 100);
        // 停止旧系统
        break;
    }
    
    console.log(`迁移到阶段: ${phase}`);
  }
  
  // 6. 回滚机制

    // 禁用新系统
    this.enableNewTracking('*', 0);
    
    // 恢复旧系统
    await this.oldSystem.enable();
    
    // 记录回滚事件
    this.recordEvent('system_rollback', {

      to: 'old',
      timestamp: Date.now()
    });

  }
```

```javascript
// 1. SSE协议基础
class SSEProtocol {
  // SSE消息格式
  static formatMessage(data, options = {}) {
    const { id, event, retry } = options;
    let message = '';
    
    // 事件类型
    if (event) {
      message += `event: ${event}\n`;
    }
    
    // 数据字段（支持多行）
    const dataLines = data.split('\n');
    dataLines.forEach(line => {
      message += `data: ${line}\n`;
    });
    
    // 事件ID
    if (id) {
      message += `id: ${id}\n`;
    }
    
    // 重连时间
    if (retry) {
      message += `retry: ${retry}\n`;
    }
    
    // 消息分隔符
    message += '\n';
    
    return message;
  }
  
  // 解析SSE消息
  static parseMessage(rawMessage) {
    const lines = rawMessage.split('\n');
    const message = {
      data: '',
      event: 'message',
      id: null,
      retry: null
    };
    
    let dataLines = [];
    
    lines.forEach(line => {
      if (line.startsWith('data: ')) {
        dataLines.push(line.slice(6));
      } else if (line.startsWith('event: ')) {
        message.event = line.slice(7);
      } else if (line.startsWith('id: ')) {
        message.id = line.slice(4);
      } else if (line.startsWith('retry: ')) {
        message.retry = parseInt(line.slice(7));
      }
    });
    
    message.data = dataLines.join('\n');
    
    return message;
  }
}

// 2. 服务端实现
class SSEServer {
  constructor() {
    this.clients = new Map();
    this.eventHistory = [];
  }
  
  // 处理SSE连接
  handleConnection(req, res) {
    // 设置SSE响应头
    res.writeHead(200, {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive',
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Headers': 'Cache-Control'
    });
    
    // 生成客户端ID
    const clientId = this.generateClientId();
    
    // 存储客户端连接
    this.clients.set(clientId, {
      id: clientId,
      response: res,
      connectedAt: Date.now(),
      lastEventId: null
    });
    
    // 发送初始连接事件
    this.sendToClient(clientId, {
      event: 'connected',
      data: JSON.stringify({ clientId })
    });
    
    // 发送历史事件（如果客户端重连）
    const lastEventId = req.headers['last-event-id'];
    if (lastEventId) {
      this.sendMissedEvents(clientId, lastEventId);
    }
    
    // 心跳检测
    const heartbeatInterval = setInterval(() => {
      this.sendToClient(clientId, { event: 'heartbeat', data: '' });
    }, 30000);
    
    // 客户端断开连接
    req.on('close', () => {
      clearInterval(heartbeatInterval);
      this.clients.delete(clientId);
      console.log(`客户端 ${clientId} 断开连接`);
    });
    
    console.log(`客户端 ${clientId} 已连接`);
  }
  
  // 发送事件到客户端
  sendToClient(clientId, event) {
    const client = this.clients.get(clientId);
    
    if (!client) {
      console.warn(`客户端 ${clientId} 不存在`);
      return false;
    }
    
    try {
      const message = SSEProtocol.formatMessage(event.data, {
        event: event.event,
        id: event.id || Date.now().toString(),
        retry: event.retry
      });
      
      client.response.write(message);
      
      // 记录事件历史
      this.eventHistory.push({
        ...event,
        clientId,
        timestamp: Date.now()
      });
      
      // 限制历史记录大小
      if (this.eventHistory.length > 1000) {
        this.eventHistory = this.eventHistory.slice(-500);
      }
      
      return true;
    } catch (error) {
      console.error(`发送事件到客户端 ${clientId} 失败:`, error);
      this.clients.delete(clientId);
      return false;
    }
  }
  
  // 广播事件到所有客户端
  broadcast(event) {
    let successCount = 0;
    
    this.clients.forEach((client, clientId) => {
      if (this.sendToClient(clientId, event)) {
        successCount++;
      }
    });
    
    return successCount;
  }
  
  // 发送错过的事件（重连时）
  sendMissedEvents(clientId, lastEventId) {
    const missedEvents = this.eventHistory.filter(event => 
      event.id > lastEventId
    );
    
    missedEvents.forEach(event => {
      this.sendToClient(clientId, event);
    });
    
    console.log(`向客户端 ${clientId} 发送了 ${missedEvents.length} 个错过的事件`);
  }
  
  generateClientId() {
    return `client-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }
}

class SSEClient {
  constructor(url, options = {}) {
    this.url = url;
    this.options = {
      withCredentials: false,
      reconnectInterval: 3000,
      maxReconnectAttempts: 5,
      ...options
    };
    
    this.eventSource = null;
    this.reconnectAttempts = 0;
    this.listeners = new Map();
    this.lastEventId = null;
    this.state = 'disconnected'; // disconnected, connecting, connected
  }
  
  // 连接
  connect() {
    if (this.state === 'connecting' || this.state === 'connected') {
      console.warn('已经连接或正在连接');
      return;
    }
    
    this.state = 'connecting';
    
    // 构建URL
    const url = new URL(this.url);
    if (this.lastEventId) {
      url.searchParams.set('lastEventId', this.lastEventId);
    }
    
    // 创建EventSource
    this.eventSource = new EventSource(url.toString(), {
      withCredentials: this.options.withCredentials
    });
    
    // 连接成功
    this.eventSource.onopen = () => {
      console.log('SSE连接已建立');
      this.state = 'connected';
      this.reconnectAttempts = 0;
      this.emit('open');
    };
    
    // 接收消息
    this.eventSource.onmessage = (event) => {
      this.lastEventId = event.lastEventId;
      this.handleMessage(event);
    };
    
    // 连接错误
    this.eventSource.onerror = (error) => {
      console.error('SSE连接错误:', error);
      this.state = 'disconnected';
      this.emit('error', error);
      this.reconnect();
    };
    
    // 自定义事件监听
    this.eventSource.addEventListener('heartbeat', () => {
      this.emit('heartbeat');
    });
    
    this.eventSource.addEventListener('connected', (event) => {
      const data = JSON.parse(event.data);
      this.emit('connected', data);
    });
  }
  
  // 处理消息
  handleMessage(event) {
    try {
      const data = JSON.parse(event.data);
      this.emit('message', data);
    } catch (error) {
      console.error('解析消息失败:', error);
      this.emit('error', error);
    }
  }
  
  // 重连逻辑
  reconnect() {
    if (this.reconnectAttempts >= this.options.maxReconnectAttempts) {
      console.error('达到最大重连次数');
      this.emit('maxReconnectAttemptsReached');
      return;
    }
    
    this.reconnectAttempts++;
    
    // 指数退避算法
    const delay = this.options.reconnectInterval * 
      Math.pow(2, this.reconnectAttempts - 1);
    
    console.log(`${delay}ms 后尝试重连 (${this.reconnectAttempts}/${this.options.maxReconnectAttempts})`);
    
    setTimeout(() => {
      this.connect();
    }, delay);
  }
  
  // 事件监听
  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
    
    return () => this.off(event, callback);
  }
  
  off(event, callback) {
    const callbacks = this.listeners.get(event) || [];
    const index = callbacks.indexOf(callback);
    if (index > -1) {
      callbacks.splice(index, 1);
    }
  }
  
  emit(event, data) {
    const callbacks = this.listeners.get(event) || [];
    callbacks.forEach(callback => {
      try {
        callback(data);
      } catch (error) {
        console.error(`事件 ${event} 处理失败:`, error);
      }
    });
  }
  
  // 关闭连接
  close() {
    if (this.eventSource) {
      this.eventSource.close();
      this.eventSource = null;
    }
    this.state = 'disconnected';
    this.emit('close');
  }
  
  // 发送消息（通过HTTP POST）
  async send(message) {
    const response = await fetch(this.url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(message)
    });
    
    return response.json();
  }
  
  // 获取连接状态
  getState() {
    return {
      state: this.state,
      reconnectAttempts: this.reconnectAttempts,
      lastEventId: this.lastEventId
    };
  }
}

// 4. 流式对话实现
class StreamingChat {
  constructor(sseUrl) {
    this.sseClient = new SSEClient(sseUrl);
    this.messageBuffer = '';
    this.isStreaming = false;
    
    this.setupEventListeners();
  }
  
  setupEventListeners() {
    this.sseClient.on('message', (data) => {
      this.handleStreamMessage(data);
    });
    
    this.sseClient.on('connected', (data) => {
      console.log('聊天连接已建立:', data);
    });
    
    this.sseClient.on('error', (error) => {
      console.error('聊天连接错误:', error);
    });
  }
  
  // 开始流式对话
  async startChat(message) {
    this.isStreaming = true;
    this.messageBuffer = '';
    
    // 发送消息
    await this.sseClient.send({
      action: 'chat',
      message: message
    });
    
    // 连接SSE
    this.sseClient.connect();
    
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        reject(new Error('聊天超时'));
      }, 60000);
      
      this.sseClient.on('done', () => {
        clearTimeout(timeout);
        this.isStreaming = false;
        resolve(this.messageBuffer);
      });
      
      this.sseClient.on('error', (error) => {
        clearTimeout(timeout);
        this.isStreaming = false;
        reject(error);
      });
    });
  }
  
  // 处理流式消息
  handleStreamMessage(data) {
    if (data.type === 'chunk') {
      // 增量更新
      this.messageBuffer += data.content;
      this.emit('chunk', data.content);
    } else if (data.type === 'done') {
      // 完成
      this.emit('done', this.messageBuffer);
    } else if (data.type === 'error') {
      // 错误
      this.emit('error', data.error);
    }
  }
  
  // 停止流式对话
  stop() {
    this.sseClient.send({ action: 'stop' });
    this.isStreaming = false;
  }
  
  // 事件监听
  on(event, callback) {
    this.sseClient.on(event, callback);
  }
  
  emit(event, data) {
    this.sseClient.emit(event, data);
  }
}
```
