class IntentClassifier {
  constructor() {
    this.intents = {
      'product_search': ['找', '搜索', '查询', '有没有'],
      'price_inquiry': ['多少钱', '价格', '便宜', '贵'],
      'comparison': ['对比', '比较', '哪个好', '区别'],
      'review': ['评价', '怎么样', '好不好', '口碑'],
      'recommendation': ['推荐', '建议', '适合', '送礼']
    };
  }
  
  async classify(text) {
    const scores = {};
    
    for (const [intent, keywords] of Object.entries(this.intents)) {
      scores[intent] = 0;
      
      for (const keyword of keywords) {
        if (text.includes(keyword)) {
          scores[intent] += 1;
        }
      }
    }
    
    // 选择得分最高的意图
    const maxScore = Math.max(...Object.values(scores));
    const topIntent = Object.keys(scores).find(key => scores[key] === maxScore);
    
    return {
      intent: topIntent,
      confidence: maxScore / 3, // 归一化
      allScores: scores
    };
  }
}

class OptimizedVectorStore {
  constructor() {
    this.index = null;
    this.dimension = 768;
    this.documents = new Map();
  }
  
  // 构建HNSW索引
  async buildIndex(documents) {
    // 使用HNSW算法构建近似最近邻索引
    this.index = new HNSWIndex(this.dimension);
    
    for (const doc of documents) {
      const embedding = await this.getEmbedding(doc.content);
      this.index.addItem(embedding, doc.id);
      this.documents.set(doc.id, doc);
    }
    
    this.index.buildIndex();
  }
  
  // 混合检索
  async hybridSearch(query, options = {}) {
    const { topK = 10, alpha = 0.5 } = options;
    
    // 语义检索
    const queryEmbedding = await this.getEmbedding(query);
    const semanticResults = this.index.search(queryEmbedding, topK * 2);
    
    // 关键词检索
    const keywordResults = this.keywordSearch(query, topK * 2);
    
    // 混合排序
    const mergedResults = this.mergeResults(
      semanticResults,
      keywordResults,
      alpha
    );
    
    return mergedResults.slice(0, topK);
  }
  
  // 关键词检索
  keywordSearch(query, topK) {
    const keywords = this.extractKeywords(query);
    const results = [];
    
    for (const [id, doc] of this.documents) {
      let score = 0;
      
      for (const keyword of keywords) {
        if (doc.content.includes(keyword)) {
          score += 1;
        }
      }
      
      if (score > 0) {
        results.push({ id, score, document: doc });
      }
    }
    
    return results
      .sort((a, b) => b.score - a.score)
      .slice(0, topK);
  }
  
  // 合并结果
  mergeResults(semanticResults, keywordResults, alpha) {
    const resultMap = new Map();
    
    // 语义检索结果
    for (const result of semanticResults) {
      resultMap.set(result.id, {
        ...result,
        semanticScore: result.score,
        keywordScore: 0
      });
    }
    
    // 关键词检索结果
    for (const result of keywordResults) {
      if (resultMap.has(result.id)) {
        resultMap.get(result.id).keywordScore = result.score;
      } else {
        resultMap.set(result.id, {
          ...result,
          semanticScore: 0,
          keywordScore: result.score
        });
      }
    }
    
    // 计算混合分数
    const results = Array.from(resultMap.values());
    
    for (const result of results) {
      result.score = alpha * result.semanticScore + 
                     (1 - alpha) * result.keywordScore;
    }
    
    return results.sort((a, b) => b.score - a.score);
  }
  
  async getEmbedding(text) {
    // 调用向量化API
    const response = await fetch('/api/embedding', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text })
    });
    
    const data = await response.json();
    return data.embedding;
  }
  
  extractKeywords(text) {
    // 简单的关键词提取
    return text
      .replace(/[^\w\s\u4e00-\u9fa5]/g, '')
      .split(/\s+/)
      .filter(word => word.length > 1);
  }
}

// 4. 流式回答生成
class StreamingAnswerGenerator {
  constructor() {
    this.abortController = null;
  }
  
  async generateStream(question, context, onChunk, onComplete) {
    this.abortController = new AbortController();
    
    try {
      const response = await fetch('/api/generate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ question, context }),
        signal: this.abortController.signal
      });
      
      const reader = response.body.getReader();
      const decoder = new TextDecoder();
      let fullAnswer = '';
      
      while (true) {
        const { done, value } = await reader.read();
        
        if (done) {
          onComplete(fullAnswer);
          break;
        }
        
        const chunk = decoder.decode(value, { stream: true });
        fullAnswer += chunk;
        
        onChunk(chunk, fullAnswer);
      }
      
    } catch (error) {
      if (error.name === 'AbortError') {
        console.log('生成已取消');
      } else {
        throw error;
      }
    }
  }
  
  stop() {
    if (this.abortController) {
      this.abortController.abort();
    }
  }
}

// 1. Redis向量存储
class RedisVectorStore {
  constructor(redisClient) {
    this.redis = redisClient;
    this.indexName = 'vector_index';
  }
  
  // 创建向量索引
  async createIndex(dimension, algorithm = 'HNSW') {
    const schema = {
      '$.id': {
        type: 'TAG',
        AS: 'id'
      },
      '$.content': {
        type: 'TEXT',
        AS: 'content'
      },
      '$.embedding': {
        type: 'VECTOR',
        AS: 'embedding',
        ALGORITHM: algorithm,
        DIMENSION: dimension,
        DISTANCE_METRIC: 'COSINE'
      }
    };
    
    try {
      await this.redis.call(
        'FT.CREATE',
        this.indexName,
        'ON', 'JSON',
        'PREFIX', '1', 'doc:',
        'SCHEMA',
        ...Object.entries(schema).flatMap(([field, config]) => [
          field,
          ...Object.entries(config).flatMap(([key, value]) => [key, value])
        ])
      );
      
      console.log('索引创建成功');
    } catch (error) {
      if (error.message.includes('Index already exists')) {
        console.log('索引已存在');
      } else {
        throw error;
      }
    }
  }
  
  // 添加文档
  async addDocument(id, content, embedding) {
    const docKey = `doc:${id}`;
    
    await this.redis.call(
      'JSON.SET',
      docKey,
      '$',
      JSON.stringify({
        id,
        content,
        embedding
      })
    );
  }
  
  // 批量添加文档
  async addDocuments(documents) {
    const pipeline = this.redis.pipeline();
    
    for (const doc of documents) {
      const docKey = `doc:${doc.id}`;
      
      pipeline.call(
        'JSON.SET',
        docKey,
        '$',
        JSON.stringify({
          id: doc.id,
          content: doc.content,
          embedding: doc.embedding
        })
      );
    }
    
    await pipeline.exec();
  }
  
  // 向量检索
  async search(queryEmbedding, topK = 10, filters = {}) {
    let query = `*=>[KNN ${topK} @embedding $query_vector AS score]`;
    
    // 添加过滤条件
    const filterConditions = [];
    
    if (filters.category) {
      filterConditions.push(`@category:{${filters.category}}`);
    }
    
    if (filters.priceRange) {
      filterConditions.push(`@price:[${filters.priceRange.min} ${filters.priceRange.max}]`);
    }
    
    if (filterConditions.length > 0) {
      query = `(${filterConditions.join(' ')})=>[KNN ${topK} @embedding $query_vector AS score]`;
    }
    
    const results = await this.redis.call(
      'FT.SEARCH',
      this.indexName,
      query,
      'PARAMS', '2', 'query_vector', Buffer.from(new Float32Array(queryEmbedding).buffer),
      'SORTBY', 'score',
      'DIALECT', '2'
    );
    
    return this.parseSearchResults(results);
  }
  
  // 解析搜索结果
  parseSearchResults(results) {
    const [total, ...docs] = results;
    const parsed = [];
    
    for (let i = 0; i < docs.length; i += 2) {
      const key = docs[i];
      const fields = docs[i + 1];
      
      const doc = {};
      for (let j = 0; j < fields.length; j += 2) {
        doc[fields[j]] = fields[j + 1];
      }
      
      parsed.push({
        id: doc.id,
        content: doc.content,
        score: parseFloat(doc.score)
      });
    }

  }

// 2. HNSW算法实现
class HNSWIndex {
  constructor(dimension, M = 16, efConstruction = 200) {
    this.dimension = dimension;
    this.M = M; // 每层的最大连接数
    this.efConstruction = efConstruction; // 构建时的搜索宽度
    this.maxLevel = 0;
    this.entryPoint = null;
    this.nodes = new Map();
  }
  
  // 添加节点
  addItem(vector, id) {
    const level = this.getRandomLevel();
    const node = {
      id,
      vector,
      level,
      connections: new Map() // 每层的连接
    };
    
    // 初始化每层的连接
    for (let i = 0; i <= level; i++) {
      node.connections.set(i, []);
    }
    
    this.nodes.set(id, node);
    
    if (this.entryPoint === null || level > this.maxLevel) {
      this.entryPoint = id;
      this.maxLevel = level;
    }
    
    return node;
  }
  
  // 构建索引
  buildIndex() {
    const nodeIds = Array.from(this.nodes.keys());
    
    for (const nodeId of nodeIds) {
      this.insertNode(nodeId);
    }
  }
  
  // 插入节点
  insertNode(nodeId) {
    const node = this.nodes.get(nodeId);
    const { vector, level } = node;
    
    let currentEntryPoint = this.entryPoint;
    let currentMaxLevel = this.maxLevel;
    
    // 从顶层到底层搜索最近邻
    for (let l = currentMaxLevel; l > level; l--) {
      const nearest = this.searchLayer(vector, currentEntryPoint, 1, l)[0];
      currentEntryPoint = nearest.id;
    }
    
    // 在每一层添加连接
    for (let l = Math.min(level, currentMaxLevel); l >= 0; l--) {
      const nearestNeighbors = this.searchLayer(
        vector,
        currentEntryPoint,
        this.efConstruction,
        l
      );
      
      // 选择M个最近邻
      const selectedNeighbors = this.selectNeighbors(
        nearestNeighbors,
        this.M
      );
      
      // 添加连接
      for (const neighbor of selectedNeighbors) {
        this.addConnection(nodeId, neighbor.id, l);
        this.addConnection(neighbor.id, nodeId, l);
        
        // 维护连接数限制
        this.maintainConnections(neighbor.id, l);
      }
      
      currentEntryPoint = selectedNeighbors[0].id;
    }
  }
  
  // 搜索
  search(queryVector, k = 10) {
    if (this.entryPoint === null) {
      return [];
    }
    
    let currentEntryPoint = this.entryPoint;
    
    // 从顶层到底层搜索
    for (let l = this.maxLevel; l > 0; l--) {
      const nearest = this.searchLayer(queryVector, currentEntryPoint, 1, l)[0];
      currentEntryPoint = nearest.id;
    }
    
    // 在第0层搜索k个最近邻
    const results = this.searchLayer(queryVector, currentEntryPoint, k, 0);
    
    return results.slice(0, k);
  }
  
  // 层内搜索
  searchLayer(queryVector, entryPointId, k, layer) {
    const visited = new Set();
    const candidates = new PriorityQueue((a, b) => a.distance - b.distance);
    const results = new PriorityQueue((a, b) => a.distance - b.distance);
    
    // 计算入口点距离
    const entryPoint = this.nodes.get(entryPointId);
    const entryDistance = this.calculateDistance(queryVector, entryPoint.vector);
    
    candidates.push({ id: entryPointId, distance: entryDistance });
    results.push({ id: entryPointId, distance: entryDistance });
    visited.add(entryPointId);
    
    while (!candidates.isEmpty()) {
      const current = candidates.pop();
      const currentDist = current.distance;
      const worstResult = results.peek();
      
      if (currentDist > worstResult.distance) {
        break;
      }
      
      const currentNode = this.nodes.get(current.id);
      const connections = currentNode.connections.get(layer) || [];
      
      for (const neighborId of connections) {
        if (visited.has(neighborId)) {
          continue;
        }
        
        visited.add(neighborId);
        
        const neighbor = this.nodes.get(neighborId);
        const distance = this.calculateDistance(queryVector, neighbor.vector);
        
        if (results.size() < k || distance < worstResult.distance) {
          candidates.push({ id: neighborId, distance });
          results.push({ id: neighborId, distance });
          
          if (results.size() > k) {
            results.pop();
          }
        }
      }
    }
    
    return results.toArray();
  }
  
  // 选择邻居
  selectNeighbors(candidates, M) {
    // 简单实现：选择最近的M个
    return candidates.slice(0, M);
  }
  
  // 添加连接
  addConnection(nodeId, neighborId, layer) {
    const node = this.nodes.get(nodeId);
    const connections = node.connections.get(layer);
    
    if (!connections.includes(neighborId)) {
      connections.push(neighborId);
    }
  }
  
  // 维护连接数限制
  maintainConnections(nodeId, layer) {
    const node = this.nodes.get(nodeId);
    const connections = node.connections.get(layer);
    
    if (connections.length > this.M * 2) {
      // 修剪连接，保留最近的M个
      const neighbors = connections.map(id => {
        const neighbor = this.nodes.get(id);
        return {
          id,
          distance: this.calculateDistance(node.vector, neighbor.vector)
        };
      });
      
      neighbors.sort((a, b) => a.distance - b.distance);
      
      node.connections.set(layer, neighbors.slice(0, this.M).map(n => n.id));
    }
  }
  
  // 计算距离
  calculateDistance(vec1, vec2) {
    // 余弦距离
    let dotProduct = 0;
    let norm1 = 0;
    let norm2 = 0;
    
    for (let i = 0; i < this.dimension; i++) {
      dotProduct += vec1[i] * vec2[i];
      norm1 += vec1[i] * vec1[i];
      norm2 += vec2[i] * vec2[i];
    }
    
    return 1 - (dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2)));
  }
  
  // 获取随机层级
  getRandomLevel() {
    const mL = 1 / Math.log(this.M);
    return Math.floor(-Math.log(Math.random()) * mL);
  }
}

// 3. 优先队列实现
class PriorityQueue {
  constructor(comparator) {
    this.heap = [];
    this.comparator = comparator;
  }
  
  push(item) {
    this.heap.push(item);
    this.bubbleUp(this.heap.length - 1);
  }
  
  pop() {
    const top = this.heap[0];
    const bottom = this.heap.pop();
    
    if (this.heap.length > 0) {
      this.heap[0] = bottom;
      this.sinkDown(0);
    }
    
    return top;
  }
  
  peek() {
    return this.heap[0];
  }
  
  size() {
    return this.heap.length;
  }
  
  isEmpty() {
    return this.heap.length === 0;
  }
  
  toArray() {
    return [...this.heap].sort(this.comparator);
  }
  
  bubbleUp(index) {
    while (index > 0) {
      const parent = Math.floor((index - 1) / 2);
      
      if (this.comparator(this.heap[index], this.heap[parent]) < 0) {
        [this.heap[index], this.heap[parent]] = [this.heap[parent], this.heap[index]];
        index = parent;
      } else {
        break;
      }
    }
  }
  
  sinkDown(index) {
    const length = this.heap.length;
    
    while (true) {
      let smallest = index;
      const left = 2 * index + 1;
      const right = 2 * index + 2;
      
      if (left < length && this.comparator(this.heap[left], this.heap[smallest]) < 0) {
        smallest = left;
      }
      
      if (right < length && this.comparator(this.heap[right], this.heap[smallest]) < 0) {
        smallest = right;
      }
      
      if (smallest !== index) {
        [this.heap[index], this.heap[smallest]] = [this.heap[smallest], this.heap[index]];
        index = smallest;
      } else {
        break;
      }
    }
  }
}

    this.similarityThreshold = 0.8;
  }
  
  // 1. 幻觉检测
  async detectHallucination(answer, context, question) {
    const checks = await Promise.all([
      this.checkFactuality(answer, context),
      this.checkRelevance(answer, question),
      this.checkConsistency(answer, context),
      this.checkSpecificity(answer)
    ]);
    
    const hallucinationScore = this.calculateHallucinationScore(checks);
    
    return {
      isHallucination: hallucinationScore > 0.5,
      score: hallucinationScore,
      checks,
      suggestions: this.getSuggestions(checks)
    };
  }
  
  // 事实性检查
  async checkFactuality(answer, context) {
    const answerSentences = this.splitSentences(answer);
    const results = [];
    
    for (const sentence of answerSentences) {
      const similarity = await this.calculateSimilarity(sentence, context);
      const isFactuallySupported = similarity > this.similarityThreshold;
      
      results.push({
        sentence,
        similarity,
        isSupported: isFactuallySupported
      });
    }
    
    const supportedRatio = results.filter(r => r.isSupported).length / results.length;
    
    return {
      type: 'factuality',
      score: 1 - supportedRatio,
      details: results
    };
  }
  
  // 相关性检查
  async checkRelevance(answer, question) {
    const similarity = await this.calculateSimilarity(answer, question);
    
    return {
      type: 'relevance',
      score: 1 - similarity,
      similarity
    };
  }
  
  // 一致性检查
  async checkConsistency(answer, context) {
    // 检查答案是否与上下文矛盾
    const contradictions = await this.findContradictions(answer, context);
    
    return {
      type: 'consistency',
      score: contradictions.length > 0 ? 1 : 0,
      contradictions
    };
  }
  
  // 具体性检查
  async checkSpecificity(answer) {
    // 检查答案是否过于笼统
    const vaguePatterns = [
      /可能/,
      /也许/,
      /大概/,
      /一般来说/,
      /通常/,
      /有时候/
    ];
    
    const vagueCount = vaguePatterns.filter(pattern => pattern.test(answer)).length;
    
    return {
      type: 'specificity',
      score: Math.min(1, vagueCount / 3),
      vaguePhrases: vagueCount
    };
  }
  
  // 计算幻觉分数
  calculateHallucinationScore(checks) {
    const weights = {
      factuality: 0.4,
      relevance: 0.2,
      consistency: 0.3,
      specificity: 0.1
    };
    
    let totalScore = 0;
    
    for (const check of checks) {
      totalScore += check.score * weights[check.type];
    }
    
    return totalScore;
  }
  
  // 获取建议
  getSuggestions(checks) {
    const suggestions = [];
    
    for (const check of checks) {
      if (check.score > 0.5) {
        switch (check.type) {
          case 'factuality':
            suggestions.push('答案可能包含未经验证的信息，请参考原始文档');
            break;
          case 'relevance':
            suggestions.push('答案与问题的相关性较低，请重新提问');
            break;
          case 'consistency':
            suggestions.push('答案与已知信息存在矛盾，请核实');
            break;
          case 'specificity':
            suggestions.push('答案过于笼统，请提供更多细节');
            break;
        }
      }
    }
    
    return suggestions;
  }
  
  // 2. 幻觉处理策略
  async handleHallucination(hallucinationResult, originalAnswer) {
    if (!hallucinationResult.isHallucination) {
      return originalAnswer;
    }
    
    // 策略1：添加免责声明

    // 策略2：引用来源
    const citedAnswer = await this.addCitations(disclaimedAnswer);
    
    // 策略3：降级处理
    if (hallucinationResult.score > 0.8) {
      return this.generateSafeResponse(hallucinationResult);
    }
    
    return citedAnswer;
  }
  
  // 添加免责声明
  addDisclaimer(answer, hallucinationResult) {
    const disclaimers = {
      factuality: '请注意，以下信息可能需要进一步验证。',
      relevance: '以下回答可能与您的问题不完全相关。',
      consistency: '请注意，以下信息可能存在矛盾。',
      specificity: '以下回答较为笼统，仅供参考。'
    };
    
    const highScoreChecks = hallucinationResult.checks
      .filter(check => check.score > 0.5);
    
    if (highScoreChecks.length > 0) {
      const disclaimer = highScoreChecks
        .map(check => disclaimers[check.type])
        .join(' ');
      
      return `${disclaimer}\n\n${answer}`;
    }
    
    return answer;
  }
  
  // 添加引用
  async addCitations(answer) {
    const sentences = this.splitSentences(answer);
    const citedSentences = [];
    
    for (const sentence of sentences) {
      const sources = await this.findSources(sentence);
      
      if (sources.length > 0) {
        const citation = sources.map((s, i) => `[${i + 1}]`).join('');
        citedSentences.push(`${sentence} ${citation}`);
      } else {
        citedSentences.push(sentence);
      }
    }
    
    return citedSentences.join(' ');
  }
  
  // 生成安全响应
  generateSafeResponse(hallucinationResult) {
    const safeResponses = {
      factuality: '关于这个问题，我无法提供经过验证的信息。建议您查阅官方文档或联系客服。',
      relevance: '您的问题似乎超出了我的知识范围。请尝试重新描述您的问题，或联系人工客服。',
      consistency: '关于这个问题，存在不同的说法。建议您参考多个来源以获取准确信息。',
      specificity: '这个问题需要更具体的信息才能准确回答。请提供更多细节。'
    };
    
    const highScoreCheck = hallucinationResult.checks
      .find(check => check.score > 0.8);
    
    if (highScoreCheck) {

    }

  }
  
  // 辅助方法
  splitSentences(text) {
    return text.split(/[。！？.!?]/).filter(s => s.trim().length > 0);

  async calculateSimilarity(text1, text2) {

    const embedding1 = await this.getEmbedding(text1);

  }
  
  cosineSimilarity(vec1, vec2) {
    let dotProduct = 0;
    let norm1 = 0;
    let norm2 = 0;
    
    for (let i = 0; i < vec1.length; i++) {
      dotProduct += vec1[i] * vec2[i];
      norm1 += vec1[i] * vec1[i];
      norm2 += vec2[i] * vec2[i];
    }
    
    return dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2));
  }
