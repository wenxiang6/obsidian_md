## 1. React Fiber架构解决了什么问题？它的工作原理是怎样的？

**答案：**
**Fiber架构解决的问题：**

1. **主线程阻塞问题**：
   - React 16之前：递归调用栈，无法中断
   - 大量组件更新时阻塞主线程，导致UI卡顿
   - 无法区分高优先级和低优先级更新

2. **时间切片（Time Slicing）**：
   ```javascript
   // Fiber节点结构
   class FiberNode {
     constructor(tag, pendingProps, key) {
       // 节点类型
       this.tag = tag; // FunctionComponent, ClassComponent, HostComponent等
       this.key = key;
       this.elementType = null;
       this.type = null;
       this.stateNode = null;
       
       // Fiber树结构
       this.return = null; // 父Fiber
       this.child = null; // 第一个子Fiber
       this.sibling = null; // 下一个兄弟Fiber
       this.index = 0;
       
       // 状态
       this.pendingProps = pendingProps;
       this.memoizedProps = null;
       this.memoizedState = null;
       this.updateQueue = null;
       
       // 副作用
       this.flags = NoFlags;
       this.subtreeFlags = NoFlags;
       this.deletions = null;
       
       // 调度优先级
       this.lanes = NoLanes;
       this.childLanes = NoLanes;
     }
   }
   ```

3. **工作原理**：
   ```javascript
   // 双缓冲机制
   let workInProgressRoot = null; // 正在构建的Fiber树
   let currentRoot = null; // 当前显示的Fiber树
   
   // 工作循环
   function workLoop(deadline) {
     while (workInProgressRoot !== null && !shouldYield()) {
       performUnitOfWork(workInProgressRoot);
     }
     
     // 时间切片：让出主线程
     if (workInProgressRoot !== null) {
       requestIdleCallback(workLoop);
     } else {
       // 提交阶段
       commitRoot();
     }
   }
   
   // 执行单个工作单元
   function performUnitOfWork(fiber) {
     // 1. 处理当前Fiber节点
     const next = beginWork(fiber);
     
     // 2. 移动到下一个工作单元
     if (next === null) {
       // 没有子节点，完成当前节点
       completeUnitOfWork(fiber);
     } else {
       // 有子节点，继续处理
       workInProgressRoot = next;
     }
   }
   
   // 时间切片判断
   function shouldYield() {
     const deadline = performance.now();
     return deadline.timeRemaining() < 1; // 剩余时间小于1ms
   }
   ```

4. **优先级调度**：
   ```javascript
   // 优先级定义
   const PriorityLevel = {
     ImmediatePriority: 1, // 立即执行
     UserBlockingPriority: 2, // 用户阻塞
     NormalPriority: 3, // 正常
     LowPriority: 4, // 低优先级
     IdlePriority: 5 // 空闲
   };
   
   // 调度器
   class Scheduler {
     constructor() {
       this.taskQueue = [];
       this.timerQueue = [];
       this.isPerformingWork = false;
     }
     
     scheduleCallback(priorityLevel, callback) {
       const startTime = performance.now();
       const timeout = this.getTimeout(priorityLevel);
       
       const newTask = {
         id: this.nextTaskId++,
         callback,
         priorityLevel,
         startTime,
         expirationTime: startTime + timeout,
         sortIndex: startTime + timeout
       };
       
       this.push(this.taskQueue, newTask);
       
       // 请求工作
       this.requestWork();
     }
     
     getTimeout(priorityLevel) {
       switch (priorityLevel) {
         case PriorityLevel.ImmediatePriority: return 0;
         case PriorityLevel.UserBlockingPriority: return 250;
         case PriorityLevel.NormalPriority: return 5000;
         case PriorityLevel.LowPriority: return 10000;
         case PriorityLevel.IdlePriority: return 1073741823;
       }
     }
   }
   ```

// 1. 基础打字机效果
class TypewriterEffect {
  constructor(options = {}) {
    this.speed = options.speed || 50; // 每个字符的延迟（ms）
    this.cursor = options.cursor || '|';
    this.onUpdate = options.onUpdate || (() => {});
    this.onComplete = options.onComplete || (() => {});
    
    this.text = '';
    this.currentIndex = 0;
    this.isRunning = false;
    this.timer = null;
  }
  
  // 开始打字
  start(text) {
    this.text = text;
    this.currentIndex = 0;
    this.isRunning = true;
    
    this.typeNextCharacter();
  }
  
  // 打下一个字符
  typeNextCharacter() {
    if (!this.isRunning || this.currentIndex >= this.text.length) {
      this.complete();
      return;
    }
    
    const currentText = this.text.substring(0, this.currentIndex + 1);
    this.currentIndex++;
    
    // 更新显示
    this.onUpdate(currentText + this.cursor);
    
    // 计算下一个字符的延迟
    const delay = this.calculateDelay();
    
    this.timer = setTimeout(() => {
      this.typeNextCharacter();
    }, delay);
  }
  
  // 计算延迟（支持变量速度）
  calculateDelay() {
    const char = this.text[this.currentIndex];
    
    // 标点符号后延迟更长
    if ('.!?。！？'.includes(char)) {
      return this.speed * 3;
    }
    
    // 逗号后延迟
    if (',，、'.includes(char)) {
      return this.speed * 2;
    }
    
    // 换行后延迟
    if (char === '\n') {
      return this.speed * 2;
    }
    
    return this.speed;
  }
  
  // 完成打字
  complete() {
    this.isRunning = false;
    if (this.timer) {
      clearTimeout(this.timer);
    }
    this.onUpdate(this.text);
    this.onComplete();
  }
  
  // 停止打字
  stop() {
    this.isRunning = false;
    if (this.timer) {
      clearTimeout(this.timer);
    }
  }
  
  // 跳过动画，直接显示全部文本
  skip() {
    this.stop();
    this.onUpdate(this.text);
    this.onComplete();
  }
}

const TypewriterText = ({ 
  text, 
  speed = 50, 
  onComplete,
  showCursor = true,
  cursorChar = '|'
}) => {
  const [displayText, setDisplayText] = useState('');
  const [isComplete, setIsComplete] = useState(false);
  const [isTyping, setIsTyping] = useState(false);
  const indexRef = useRef(0);
  const timerRef = useRef(null);
  
  // 开始打字效果
  useEffect(() => {
    if (!text) return;
    
    indexRef.current = 0;
    setDisplayText('');
    setIsComplete(false);
    setIsTyping(true);
    
    const typeNext = () => {
      if (indexRef.current >= text.length) {
        setIsComplete(true);
        setIsTyping(false);
        onComplete?.();
        return;
      }
      
      indexRef.current++;
      setDisplayText(text.substring(0, indexRef.current));
      
      // 计算延迟
      const char = text[indexRef.current - 1];
      let delay = speed;
      
      if ('.!?。！？'.includes(char)) {
        delay = speed * 3;
      } else if (',，、'.includes(char)) {
        delay = speed * 2;
      } else if (char === '\n') {
        delay = speed * 2;
      }
      
      timerRef.current = setTimeout(typeNext, delay);
    };
    
    timerRef.current = setTimeout(typeNext, speed);
    
    return () => {
      if (timerRef.current) {
        clearTimeout(timerRef.current);
      }
    };
  }, [text, speed, onComplete]);
  
  // 跳过动画
  const handleSkip = useCallback(() => {
    if (timerRef.current) {
      clearTimeout(timerRef.current);
    }
    setDisplayText(text);
    setIsComplete(true);
    setIsTyping(false);
    onComplete?.();
  }, [text, onComplete]);
  
  return (
    <div className="typewriter-container">
      <span className="typewriter-text">
        {displayText}

      {isTyping && (

          className="skip-button"

        >
          跳过
        </button>
      )}
    </div>
  );
};

class StreamingTypewriter {
  constructor(options = {}) {
    this.speed = options.speed || 30;
    this.buffer = '';
    this.displayText = '';
    this.isProcessing = false;
    this.onUpdate = options.onUpdate || (() => {});
    this.onComplete = options.onComplete || (() => {});
    
    this.processQueue = this.processQueue.bind(this);
  }
  
  // 添加新的数据块
  addChunk(chunk) {
    this.buffer += chunk;
    
    if (!this.isProcessing) {
      this.processQueue();
    }
  }
  
  // 处理队列
  async processQueue() {
    this.isProcessing = true;
    
    while (this.buffer.length > 0) {
      const char = this.buffer[0];
      this.buffer = this.buffer.substring(1);
      
      this.displayText += char;
      this.onUpdate(this.displayText);
      
      // 计算延迟
      const delay = this.calculateDelay(char);
      await this.sleep(delay);
    }
    
    this.isProcessing = false;
    this.onComplete(this.displayText);
  }
  
  calculateDelay(char) {
    if ('.!?。！？'.includes(char)) {
      return this.speed * 3;
    }
    if (',，、'.includes(char)) {
      return this.speed * 2;
    }
    if (char === '\n') {
      return this.speed * 2;
    }
    return this.speed;
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
  
  // 清空缓冲区
  clear() {
    this.buffer = '';
    this.displayText = '';
    this.isProcessing = false;
  }
}

// 5. 性能优化版本
const OptimizedTypewriter = ({ text, speed = 30 }) => {
  const [displayText, setDisplayText] = useState('');
  const requestRef = useRef();
  const previousTimeRef = useRef();
  const indexRef = useRef(0);
  
  useEffect(() => {
    indexRef.current = 0;
    setDisplayText('');
    
    const animate = (time) => {
      if (previousTimeRef.current !== undefined) {
        const deltaTime = time - previousTimeRef.current;
        
        // 根据时间差计算应该显示的字符数
        const charsToAdd = Math.floor(deltaTime / speed);
        
        if (charsToAdd > 0 && indexRef.current < text.length) {
          const newIndex = Math.min(
            indexRef.current + charsToAdd,
            text.length
          );
          
          setDisplayText(text.substring(0, newIndex));
          indexRef.current = newIndex;
        }
      }
      
      previousTimeRef.current = time;
      
      if (indexRef.current < text.length) {
        requestRef.current = requestAnimationFrame(animate);
      }
    };
    
    requestRef.current = requestAnimationFrame(animate);
    
    return () => {
      if (requestRef.current) {
        cancelAnimationFrame(requestRef.current);
      }
    };
  }, [text, speed]);
  
  return (
    <div>
      {displayText}
      {indexRef.current < text.length && <span className="cursor">|</span>}
    </div>
  );
};

// 4. Markdown解析器
class MarkdownParser {
  constructor() {
    this.rules = [
      // 标题
      { pattern: /^#{1,6}\s+.+$/gm, type: 'heading' },
      // 代码块
      { pattern: /```[\s\S]*?```/g, type: 'codeblock' },
      // 行内代码
      { pattern: /`[^`]+`/g, type: 'inlinecode' },
      // 数学公式块
      { pattern: /\$\$[\s\S]*?\$\$/g, type: 'mathblock' },
      // 行内数学公式
      { pattern: /\$[^$]+\$/g, type: 'inlinemath' },
      // 链接
      { pattern: /\[([^\]]+)\]\(([^)]+)\)/g, type: 'link' },
      // 图片
      { pattern: /!\[([^\]]*)\]\(([^)]+)\)/g, type: 'image' },
      // 列表
      { pattern: /^[\s]*[-*+]\s+.+$/gm, type: 'list' },
      // 引用
      { pattern: /^>\s+.+$/gm, type: 'blockquote' }
    ];
  }
  
  parse(markdown) {
    let result = markdown;
    
    // 处理数学公式（最先处理，避免被其他规则干扰）
    result = this.processMathBlocks(result);
    result = this.processInlineMath(result);
    
    // 处理代码块
    result = this.processCodeBlocks(result);
    result = this.processInlineCode(result);
    
    // 处理其他元素
    result = this.processHeadings(result);
    result = this.processLinks(result);
    result = this.processImages(result);
    result = this.processLists(result);
    result = this.processBlockquotes(result);
    
    // 处理段落
    result = this.processParagraphs(result);
    
    return result;
  }
  
  processMathBlocks(text) {
    return text.replace(/\$\$([\s\S]*?)\$\$/g, (match, formula) => {
      return `<div class="math-block" data-formula="${encodeURIComponent(formula)}"></div>`;
    });
  }
  
  processInlineMath(text) {
    return text.replace(/\$([^$]+)\$/g, (match, formula) => {
      return `<span class="math-inline" data-formula="${encodeURIComponent(formula)}"></span>`;
    });
  }
  
  processCodeBlocks(text) {
    return text.replace(/```(\w+)?\n([\s\S]*?)```/g, (match, lang, code) => {
      return `<pre><code class="language-${lang || 'text'}">${this.escapeHtml(code.trim())}</code></pre>`;
    });
  }
  
  processInlineCode(text) {
    return text.replace(/`([^`]+)`/g, (match, code) => {
      return `<code>${this.escapeHtml(code)}</code>`;
    });
  }
  
  processHeadings(text) {
    return text.replace(/^(#{1,6})\s+(.+)$/gm, (match, hashes, content) => {
      const level = hashes.length;
      return `<h${level}>${content}</h${level}>`;
    });
  }
  
  processLinks(text) {
    return text.replace(/\[([^\]]+)\]\(([^)]+)\)/g, (match, text, url) => {
      return `<a href="${url}" target="_blank" rel="noopener noreferrer">${text}</a>`;
    });
  }
  
  processImages(text) {
    return text.replace(/!\[([^\]]*)\]\(([^)]+)\)/g, (match, alt, src) => {
      return `<img src="${src}" alt="${alt}" loading="lazy" />`;
    });
  }
  
  processLists(text) {
    // 处理无序列表
    text = text.replace(/^[\s]*[-*+]\s+(.+)$/gm, '<li>$1</li>');
    text = text.replace(/(<li>.*<\/li>\n?)+/g, '<ul>$&</ul>');
    
    return text;
  }
  
  processBlockquotes(text) {
    return text.replace(/^>\s+(.+)$/gm, '<blockquote>$1</blockquote>');
  }
  
  processParagraphs(text) {
    // 将连续的文本包装在<p>标签中
    return text.replace(/^(?!<[a-z])((?!<).+)$/gm, '<p>$1</p>');
  }
  
  escapeHtml(text) {
    const map = {
      '&': '&amp;',
      '<': '&lt;',
      '>': '&gt;',
      '"': '&quot;',
      "'": '&#039;'
    };
    return text.replace(/[&<>"']/g, m => map[m]);
  }
}

class LaTeXRenderer {
  constructor() {
    this.katex = null;
    this.loadKaTeX();
  }
  
  async loadKaTeX() {
    try {
      const katex = await import('katex');
      this.katex = katex.default;
    } catch (error) {
      console.error('加载KaTeX失败:', error);
    }
  }
  
  renderMath(formula, displayMode = false) {
    if (!this.katex) {
      return formula;
    }
    
    try {
      return this.katex.renderToString(formula, {
        throwOnError: false,
        displayMode: displayMode
      });
    } catch (error) {
      console.error('LaTeX渲染失败:', error);
      return formula;
    }
  }
  
  renderElement(element) {
    const formula = decodeURIComponent(element.dataset.formula);
    const isDisplay = element.classList.contains('math-block');
    
    const rendered = this.renderMath(formula, isDisplay);
    element.innerHTML = rendered;
  }
  
  renderAll(container) {
    const mathElements = container.querySelectorAll('.math-block, .math-inline');
    mathElements.forEach(element => this.renderElement(element));
  }
}

// 1. 性能监控系统
class PerformanceMonitor {
  constructor() {
    this.metrics = {};
    this.observers = [];
    this.init();
  }
  
  init() {
    // 监控Core Web Vitals
    this.observeLCP();
    this.observeFID();
    this.observeCLS();
    
    // 监控其他指标
    this.observeFCP();
    this.observeTTI();
    this.observeTTFB();
  }
  
  // Largest Contentful Paint (LCP)
  observeLCP() {
    const observer = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      const lastEntry = entries[entries.length - 1];
      
      this.metrics.lcp = {
        value: lastEntry.startTime,
        element: lastEntry.element,
        url: lastEntry.url,
        timestamp: Date.now()
      };
      
      this.reportMetric('LCP', this.metrics.lcp);
    });
    
    observer.observe({ type: 'largest-contentful-paint', buffered: true });
    this.observers.push(observer);
  }
  
  // First Input Delay (FID)
  observeFID() {
    const observer = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      entries.forEach(entry => {
        this.metrics.fid = {
          value: entry.processingStart - entry.startTime,
          event: entry.name,
          timestamp: Date.now()
        };
        
        this.reportMetric('FID', this.metrics.fid);
      });
    });
    
    observer.observe({ type: 'first-input', buffered: true });
    this.observers.push(observer);
  }
  
  // Cumulative Layout Shift (CLS)
  observeCLS() {
    let clsValue = 0;
    let clsEntries = [];
    
    const observer = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      entries.forEach(entry => {
        if (!entry.hadRecentInput) {
          clsValue += entry.value;
          clsEntries.push(entry);
          
          this.metrics.cls = {
            value: clsValue,
            entries: clsEntries,
            timestamp: Date.now()
          };
          
          this.reportMetric('CLS', this.metrics.cls);
        }
      });
    });
    
    observer.observe({ type: 'layout-shift', buffered: true });
    this.observers.push(observer);
  }
  
  // First Contentful Paint (FCP)
  observeFCP() {
    const observer = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      entries.forEach(entry => {
        if (entry.name === 'first-contentful-paint') {
          this.metrics.fcp = {
            value: entry.startTime,
            timestamp: Date.now()
          };
          
          this.reportMetric('FCP', this.metrics.fcp);
        }
      });
    });
    
    observer.observe({ type: 'paint', buffered: true });
    this.observers.push(observer);
  }
  
  // Time to Interactive (TTI)
  observeTTI() {
    // TTI计算比较复杂，这里简化实现
    const ttiPolyfill = () => {
      // 检查是否有长任务
      const longTasks = performance.getEntriesByType('longtask');
      const lastLongTask = longTasks[longTasks.length - 1];
      
      if (lastLongTask) {
        this.metrics.tti = {
          value: lastLongTask.startTime + lastLongTask.duration,
          timestamp: Date.now()
        };
      }
    };
    
    // 在页面加载完成后计算
    window.addEventListener('load', () => {
      setTimeout(ttiPolyfill, 0);
    });
  }
  
  // Time to First Byte (TTFB)
  observeTTFB() {
    const navigationEntry = performance.getEntriesByType('navigation')[0];
    
    if (navigationEntry) {
      this.metrics.ttfb = {
        value: navigationEntry.responseStart - navigationEntry.requestStart,
        timestamp: Date.now()
      };
      
      this.reportMetric('TTFB', this.metrics.ttfb);
    }
  }
  
  // 上报指标
  reportMetric(name, metric) {
    console.log(`性能指标 ${name}:`, metric);
    
    // 发送到监控系统
    this.sendToAnalytics(name, metric);
  }
  
  sendToAnalytics(name, metric) {
    // 使用sendBeacon发送
    const data = {
      metric: name,
      value: metric.value,
      url: window.location.href,
      userAgent: navigator.userAgent,
      timestamp: metric.timestamp
    };
    
    navigator.sendBeacon('/api/analytics/performance', JSON.stringify(data));
  }
  
  // 获取所有指标
  getMetrics() {
    return { ...this.metrics };
  }
  
  // 清理观察者
  disconnect() {
    this.observers.forEach(observer => observer.disconnect());
    this.observers = [];
  }
}

    this.cache.clear();
  }

class BundleAnalyzer {
  constructor(stats) {
    this.stats = stats;
  }
  
  // 分析打包结果
  analyze() {
    const analysis = {
      totalSize: this.calculateTotalSize(),
      chunks: this.analyzeChunks(),
      modules: this.analyzeModules(),
      duplicates: this.findDuplicates(),
      largeModules: this.findLargeModules()
    };
    
    return analysis;
  }
  
  calculateTotalSize() {
    return this.stats.assets.reduce((total, asset) => total + asset.size, 0);
  }
  
  analyzeChunks() {
    return this.stats.chunks.map(chunk => ({
      name: chunk.name,
      size: chunk.size,
      modules: chunk.modules.length
    }));
  }
  
  analyzeModules() {
    return this.stats.modules
      .map(module => ({
        name: module.name,
        size: module.size,
        chunks: module.chunks.length
      }))
      .sort((a, b) => b.size - a.size);
  }
  
  findDuplicates() {
    const moduleMap = new Map();
    
    this.stats.modules.forEach(module => {
      const name = module.name.split('/').pop();
      if (!moduleMap.has(name)) {
        moduleMap.set(name, []);
      }
      moduleMap.get(name).push(module);
    });
    
    return Array.from(moduleMap.entries())
      .filter(([name, modules]) => modules.length > 1)
      .map(([name, modules]) => ({
        name,
        count: modules.length,
        totalSize: modules.reduce((sum, m) => sum + m.size, 0)
      }));
  }
  
  findLargeModules(threshold = 100000) {
    return this.stats.modules
      .filter(module => module.size > threshold)
      .map(module => ({
        name: module.name,
        size: module.size,
        sizeKB: (module.size / 1024).toFixed(2) + ' KB'
      }))
      .sort((a, b) => b.size - a.size);
  }
}
