class HTTPStreaming {
  // 传统HTTP请求
  static async traditionalRequest(url) {
    // 客户端发送请求
    // 服务端处理请求
    // 服务端返回完整响应
    // 客户端接收完整响应
    const response = await fetch(url);
    const data = await response.json(); // 等待完整响应
    return data;
  }
  
  // 流式HTTP请求
  static async streamingRequest(url) {
    const response = await fetch(url);
    
    // 获取ReadableStream
    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    
    while (true) {
      const { done, value } = await reader.read();
      
      if (done) {
        console.log('流式传输完成');
        break;
      }
      
      // 处理接收到的数据块
      const chunk = decoder.decode(value, { stream: true });
      console.log('收到数据块:', chunk);
    }
  }
}

// 4. 流式数据处理
class StreamProcessor {
  constructor() {
    this.buffer = '';
    this.decoder = new TextDecoder();
  }
  
  // 处理ReadableStream
  async processStream(stream) {
    const reader = stream.getReader();
    
    try {
      while (true) {
        const { done, value } = await reader.read();
        
        if (done) {
          // 处理剩余的缓冲区数据
          if (this.buffer) {
            this.processBuffer();
          }
          break;
        }
        
        // 解码数据块
        const chunk = this.decoder.decode(value, { stream: true });
        this.buffer += chunk;
        
        // 处理完整的行
        this.processLines();
      }
    } finally {
      reader.releaseLock();
    }
  }
  
  // 处理行数据
  processLines() {
    const lines = this.buffer.split('\n');
    
    // 保留最后一行（可能不完整）
    this.buffer = lines.pop() || '';
    
    // 处理完整的行
    lines.forEach(line => {
      if (line.trim()) {
        this.processLine(line);
      }
    });
  }
  
  // 处理单行数据
  processLine(line) {
    // 解析SSE格式
    if (line.startsWith('data: ')) {
      const data = line.slice(6);
      try {
        const parsed = JSON.parse(data);
        this.handleMessage(parsed);
      } catch (e) {
        console.error('解析错误:', e);
      }
    }
  }
  
  // 处理缓冲区
  processBuffer() {
    if (this.buffer.startsWith('data: ')) {
      const data = this.buffer.slice(6);
      try {
        const parsed = JSON.parse(data);
        this.handleMessage(parsed);
      } catch (e) {
        console.error('解析错误:', e);
      }
    }
  }
  
  // 处理消息
  handleMessage(data) {
    console.log('处理消息:', data);
  }
}

class RequestManager {
  constructor() {
    this.abortControllers = new Map();
    this.pendingRequests = new Map();
  }

      this.abortControllers.delete(requestId);

    }
  }
  
  // 中断所有请求
  abortAllRequests() {
    this.abortControllers.forEach((controller, id) => {
      controller.abort();

    this.abortControllers.clear();

    } else {
      // 从服务器加载
      console.log('从服务器加载对话历史');
    }

    const requestId = this.generateRequestId();
    
    // 记录活跃请求
    if (!this.activeRequests.has(conversationId)) {
      this.activeRequests.set(conversationId, []);
    }
    this.activeRequests.get(conversationId).push(requestId);
    
    try {
      // 检查对话是否仍然是当前对话
      if (conversationId !== this.currentConversationId) {
        console.log('对话已切换，取消请求');
        return null;
      }
      
      const response = await this.makeRequest(message, requestId);
      
      // 再次检查对话是否仍然是当前对话
      if (conversationId !== this.currentConversationId) {
        console.log('对话已切换，丢弃响应');
        return null;
      }
      
      return response;
      
    } finally {
      // 移除活跃请求
      const requests = this.activeRequests.get(conversationId) || [];
      const index = requests.indexOf(requestId);
      if (index > -1) {
        requests.splice(index, 1);
      }
    }
  }

    const abortController = new AbortController();
    
    // 存储AbortController
    this.abortControllers.set(requestId, abortController);
    
    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ message, requestId }),
        signal: abortController.signal
      });

    } catch (error) {
      if (error.name === 'AbortError') {
        console.log(`请求 ${requestId} 已中断`);
        return null;
      }
      throw error;
    } finally {
      this.abortControllers.delete(requestId);
    }
  }
  
  generateRequestId() {
    return `req-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }
}

class StreamingRequestManager {
  constructor() {
    this.activeStreams = new Map();
    this.streamBuffers = new Map();
  }
  
  // 开始流式请求
  async startStream(url, onChunk, onComplete, onError) {
    const streamId = this.generateStreamId();
    const abortController = new AbortController();
    
    this.activeStreams.set(streamId, {
      abortController,
      startTime: Date.now()
    });
    
    try {
      const response = await fetch(url, {
        signal: abortController.signal
      });
      
      const reader = response.body.getReader();
      const decoder = new TextDecoder();
      let buffer = '';
      
      while (true) {
        const { done, value } = await reader.read();
        
        if (done) {
          // 处理剩余的缓冲区数据
          if (buffer) {
            this.processBuffer(buffer, onChunk);
          }
          onComplete();
          break;
        }
        
        // 解码数据块
        const chunk = decoder.decode(value, { stream: true });
        buffer += chunk;
        
        // 处理完整的行
        const lines = buffer.split('\n');
        buffer = lines.pop() || '';
        
        lines.forEach(line => {
          if (line.trim()) {
            this.processLine(line, onChunk);
          }
        });
      }
      
    } catch (error) {
      if (error.name === 'AbortError') {
        console.log(`流 ${streamId} 已中断`);
        return;
      }
      onError(error);
    } finally {
      this.activeStreams.delete(streamId);
    }
  }
  
  // 中断流
  abortStream(streamId) {
    const stream = this.activeStreams.get(streamId);
    if (stream) {
      stream.abortController.abort();
      this.activeStreams.delete(streamId);
    }
  }
  
  // 中断所有流
  abortAllStreams() {
    this.activeStreams.forEach((stream, id) => {
      stream.abortController.abort();
    });
    this.activeStreams.clear();
  }
  
  processLine(line, onChunk) {
    if (line.startsWith('data: ')) {
      const data = line.slice(6);
      if (data === '[DONE]') {
        return;
      }
      try {
        const parsed = JSON.parse(data);
        onChunk(parsed);
      } catch (e) {
        console.error('解析错误:', e);
      }
    }
  }
  
  processBuffer(buffer, onChunk) {
    if (buffer.startsWith('data: ')) {
      const data = buffer.slice(6);
      try {
        const parsed = JSON.parse(data);
        onChunk(parsed);
      } catch (e) {
        console.error('解析错误:', e);
      }
    }
  }
  
  generateStreamId() {
    return `stream-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }
}

// 1. 二进制数据类型
class BinaryDataHandler {
  // ArrayBuffer处理
  static handleArrayBuffer(buffer) {
    // 创建视图
    const view = new DataView(buffer);
    
    // 读取不同类型的数据
    const intValue = view.getInt32(0, true); // 小端序
    const floatValue = view.getFloat64(4, true);
    
    // 创建Uint8Array
    const uint8Array = new Uint8Array(buffer);
    
    return {
      intValue,
      floatValue,
      bytes: Array.from(uint8Array)
    };
  }
  
  // Blob处理
  static async handleBlob(blob) {
    // 转换为ArrayBuffer
    const arrayBuffer = await blob.arrayBuffer();
    
    // 转换为文本
    const text = await blob.text();
    
    // 创建URL
    const url = URL.createObjectURL(blob);
    
    // 读取为DataURL
    const dataUrl = await new Promise((resolve) => {
      const reader = new FileReader();
      reader.onload = () => resolve(reader.result);
      reader.readAsDataURL(blob);
    });
    
    return {
      arrayBuffer,
      text,
      url,
      dataUrl,
      size: blob.size,
      type: blob.type
    };
  }
  
  // TypedArray处理
  static handleTypedArray(typedArray) {
    // 不同类型的TypedArray
    const int8 = new Int8Array(typedArray.buffer);
    const uint8 = new Uint8Array(typedArray.buffer);
    const int16 = new Int16Array(typedArray.buffer);
    const uint16 = new Uint16Array(typedArray.buffer);
    const int32 = new Int32Array(typedArray.buffer);
    const uint32 = new Uint32Array(typedArray.buffer);
    const float32 = new Float32Array(typedArray.buffer);
    const float64 = new Float64Array(typedArray.buffer);
    
    return {
      int8,
      uint8,
      int16,
      uint16,
      int32,
      uint32,
      float32,
      float64
    };
  }
}

// 2. 流式二进制数据处理
class BinaryStreamProcessor {
  constructor() {
    this.buffer = new ArrayBuffer(0);
    this.decoder = new TextDecoder();
  }
  
  // 处理流式二进制数据
  async processBinaryStream(stream) {
    const reader = stream.getReader();
    
    try {
      while (true) {
        const { done, value } = await reader.read();
        
        if (done) {
          // 处理剩余的缓冲区
          if (this.buffer.byteLength > 0) {
            this.processBuffer();
          }
          break;
        }
        
        // value是Uint8Array
        await this.processChunk(value);
      }
    } finally {
      reader.releaseLock();
    }
  }
  
  // 处理数据块
  async processChunk(chunk) {
    // 合并到缓冲区
    const newBuffer = new ArrayBuffer(this.buffer.byteLength + chunk.byteLength);
    const newView = new Uint8Array(newBuffer);
    
    newView.set(new Uint8Array(this.buffer), 0);
    newView.set(chunk, this.buffer.byteLength);
    
    this.buffer = newBuffer;
    
    // 尝试解析数据
    await this.tryParseData();
  }
  
  // 尝试解析数据
  async tryParseData() {
    // 假设数据格式：4字节长度 + 数据内容
    while (this.buffer.byteLength >= 4) {
      const view = new DataView(this.buffer);
      const length = view.getUint32(0, true); // 小端序
      
      // 检查是否有足够的数据
      if (this.buffer.byteLength < 4 + length) {
        break;
      }
      
      // 提取数据
      const data = this.buffer.slice(4, 4 + length);
      
      // 处理数据
      await this.processData(data);
      
      // 更新缓冲区
      this.buffer = this.buffer.slice(4 + length);
    }
  }
  
  // 处理解析后的数据
  async processData(data) {
    // 根据数据类型处理
    const view = new DataView(data);
    const type = view.getUint8(0);
    
    switch (type) {
      case 1: // 文本数据
        const text = this.decoder.decode(data.slice(1));
        console.log('文本数据:', text);
        break;
        
      case 2: // 数值数据
        const value = view.getFloat64(1, true);
        console.log('数值数据:', value);
        break;
        
      case 3: // 二进制数据
        const binary = new Uint8Array(data.slice(1));
        console.log('二进制数据:', binary);
        break;
        
      default:
        console.log('未知数据类型:', type);
    }
  }
  
  // 处理缓冲区
  processBuffer() {
    console.log('处理剩余缓冲区:', this.buffer.byteLength, '字节');
  }
}

class ImageBinaryHandler {
  // 处理图像二进制数据
  static async handleImageBinary(arrayBuffer) {
    // 创建Blob
    const blob = new Blob([arrayBuffer], { type: 'image/png' });
    
    // 创建URL
    const url = URL.createObjectURL(blob);
    
    // 创建Image对象
    const img = new Image();
    
    return new Promise((resolve, reject) => {
      img.onload = () => {
        // 创建Canvas
        const canvas = document.createElement('canvas');
        canvas.width = img.width;
        canvas.height = img.height;
        
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0);
        
        // 获取像素数据
        const imageData = ctx.getImageData(0, 0, img.width, img.height);
        const pixels = imageData.data;
        
        // 转换为不同格式
        const pngUrl = canvas.toDataURL('image/png');
        const jpegUrl = canvas.toDataURL('image/jpeg', 0.8);
        
        resolve({
          width: img.width,
          height: img.height,
          pixels,
          pngUrl,
          jpegUrl,
          url
        });
      };
      
      img.onerror = reject;
      img.src = url;
    });
  }
  
  // 图像处理
  static processImage(imageData, operation) {
    const { width, height, pixels } = imageData;
    const newPixels = new Uint8ClampedArray(pixels.length);
    
    for (let i = 0; i < pixels.length; i += 4) {
      const r = pixels[i];
      const g = pixels[i + 1];
      const b = pixels[i + 2];
      const a = pixels[i + 3];
      
      switch (operation) {
        case 'grayscale':
          const gray = 0.299 * r + 0.587 * g + 0.114 * b;
          newPixels[i] = gray;
          newPixels[i + 1] = gray;
          newPixels[i + 2] = gray;
          newPixels[i + 3] = a;
          break;
          
        case 'invert':
          newPixels[i] = 255 - r;
          newPixels[i + 1] = 255 - g;
          newPixels[i + 2] = 255 - b;
          newPixels[i + 3] = a;
          break;
          
        case 'brightness':
          const factor = 1.5;
          newPixels[i] = Math.min(255, r * factor);
          newPixels[i + 1] = Math.min(255, g * factor);
          newPixels[i + 2] = Math.min(255, b * factor);
          newPixels[i + 3] = a;
          break;
      }
    }
    
    return new ImageData(newPixels, width, height);
  }
}

class AudioBinaryHandler {
  // 处理音频二进制数据
  static async handleAudioBinary(arrayBuffer) {
    // 创建AudioContext
    const audioContext = new (window.AudioContext || window.webkitAudioContext)();
    
    // 解码音频数据
    const audioBuffer = await audioContext.decodeAudioData(arrayBuffer);
    
    // 获取音频信息
    const { sampleRate, length, duration, numberOfChannels } = audioBuffer;
    
    // 获取通道数据
    const channelData = [];
    for (let i = 0; i < numberOfChannels; i++) {
      channelData.push(audioBuffer.getChannelData(i));
    }
    
    return {
      sampleRate,
      length,
      duration,
      numberOfChannels,
      channelData,
      audioBuffer

    }
  }

class BinaryProtocolHandler {
  // 解析自定义二进制协议
  static parseProtocol(buffer) {
    const view = new DataView(buffer);
    let offset = 0;
    
    // 协议头
    const magic = view.getUint32(offset, false); // 大端序
    offset += 4;
    
    const version = view.getUint8(offset);
    offset += 1;
    
    const type = view.getUint8(offset);
    offset += 1;
    
    const length = view.getUint32(offset, false);
    offset += 4;
    
    // 根据类型解析数据
    let data;
    switch (type) {
      case 1: // 文本消息
        const textBytes = new Uint8Array(buffer, offset, length);
        data = new TextDecoder().decode(textBytes);
        break;
        
      case 2: // 数值数据
        data = view.getFloat64(offset, false);
        break;
        
      case 3: // 二进制数据
        data = new Uint8Array(buffer, offset, length);
        break;
        
      case 4: // JSON数据
        const jsonBytes = new Uint8Array(buffer, offset, length);
        const jsonStr = new TextDecoder().decode(jsonBytes);
        data = JSON.parse(jsonStr);
        break;
    }
    
    return {
      magic,
      version,
      type,
      length,
      data
    };
  }
  
  // 构建二进制协议
  static buildProtocol(type, data) {
    let dataBytes;
    
    switch (type) {
      case 1: // 文本消息
        dataBytes = new TextEncoder().encode(data);
        break;
        
      case 2: // 数值数据
        const buffer = new ArrayBuffer(8);
        new DataView(buffer).setFloat64(0, data, false);
        dataBytes = new Uint8Array(buffer);
        break;
        
      case 3: // 二进制数据
        dataBytes = data;
        break;
        
      case 4: // JSON数据
        const jsonStr = JSON.stringify(data);
        dataBytes = new TextEncoder().encode(jsonStr);
        break;
    }
    
    // 构建完整消息
    const message = new ArrayBuffer(10 + dataBytes.byteLength);
    const view = new DataView(message);
    
    // 协议头
    view.setUint32(0, 0x12345678, false); // magic
    view.setUint8(4, 1); // version
    view.setUint8(5, type); // type
    view.setUint32(6, dataBytes.byteLength, false); // length
    
    // 数据
    new Uint8Array(message, 10).set(dataBytes);
    
    return message;
  }
}

    const change = (avgSecond - avgFirst) / avgFirst;

    if (change < -0.1) return 'decreasing';

  }

    const cache = new Map();

    return {
      get(key) {
        if (cache.has(key)) {
          // 更新访问顺序
          const index = accessOrder.indexOf(key);
          if (index > -1) {
            accessOrder.splice(index, 1);
          }
          accessOrder.push(key);
          
          return cache.get(key);
        }
        return null;
      },
      
      set(key, value) {
        if (cache.size >= maxSize) {

          cache.delete(oldestKey);

class NetworkOptimizer {
  // 请求合并
  static createRequestBatcher(batchDelay = 50) {
    const pendingRequests = new Map();
    let batchTimer = null;
    
    const processBatch = () => {
      const requests = Array.from(pendingRequests.entries());
      pendingRequests.clear();
      
      // 合并请求
      const batchRequest = {
        requests: requests.map(([key, { request }]) => ({
          id: key,
          ...request
        }))
      };
      
      // 发送批量请求
      fetch('/api/batch', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(batchRequest)
      })
      .then(response => response.json())
      .then(results => {
        results.forEach(result => {
          const pending = requests.find(([key]) => key === result.id);
          if (pending) {
            pending[1].resolve(result.data);
          }
        });
      })
      .catch(error => {
        requests.forEach(([key, { reject }]) => {
          reject(error);
        });
      });
    };
    
    return {
      addRequest(id, request) {
        return new Promise((resolve, reject) => {
          pendingRequests.set(id, { request, resolve, reject });
          
          if (batchTimer) {
            clearTimeout(batchTimer);
          }
          
          batchTimer = setTimeout(processBatch, batchDelay);
        });
      }
    };
  }
  
  // 预加载
  static preloadResources(urls) {
    return Promise.all(

        return new Promise((resolve, reject) => {
          const link = document.createElement('link');
          link.rel = 'preload';
          link.href = url;
          link.as = this.getResourceType(url);
          link.onload = resolve;
          link.onerror = reject;
          document.head.appendChild(link);
        });
      })
    );
  }

    if (url.match(/\.(js|mjs)$/)) return 'script';
    if (url.match(/\.css$/)) return 'style';
    if (url.match(/\.(png|jpg|jpeg|gif|webp)$/)) return 'image';
    if (url.match(/\.(woff|woff2|ttf|otf)$/)) return 'font';
    return 'fetch';
