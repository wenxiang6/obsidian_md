```javascript
class AIApplicationArchitecture {
  constructor() {
    this.layers = {
      presentation: new PresentationLayer(),
      business: new BusinessLayer(),
      ai: new AILayer(),
      data: new DataLayer()
    };
  }
  
  // 完整请求流程
  async processRequest(request) {
    // 1. 表示层：用户交互
    const userInput = await this.layers.presentation.handleInput(request);
    
    // 2. 业务层：业务逻辑处理
    const businessResult = await this.layers.business.process(userInput);
    
    // 3. AI层：AI能力调用
    const aiResult = await this.layers.ai.process(businessResult);
    
    // 4. 数据层：数据持久化
    await this.layers.data.save(aiResult);
    
    // 5. 返回结果
    return this.layers.presentation.formatResponse(aiResult);
  }
}

// 表示层
class PresentationLayer {
  // 处理用户输入
  async handleInput(request) {
    const { type, data } = request;
    
    switch (type) {
      case 'text':
        return this.processTextInput(data);
      case 'voice':
        return await this.processVoiceInput(data);
      case 'image':
        return await this.processImageInput(data);
      default:
        throw new Error(`不支持的输入类型: ${type}`);
    }
  }
  
  processTextInput(data) {
    return {
      type: 'text',
      content: data.text,
      metadata: {
        timestamp: Date.now(),
        source: 'user_input'
      }
    };
  }
  
  async processVoiceInput(data) {
    // 语音转文本
    const text = await this.speechToText(data.audio);
    
    return {
      type: 'text',
      content: text,
      metadata: {
        timestamp: Date.now(),
        source: 'voice_input',
        originalAudio: data.audio
      }
    };
  }
  
  async processImageInput(data) {
    // 图像识别
    const description = await this.imageRecognition(data.image);
    
    return {
      type: 'multimodal',
      content: description,
      image: data.image,
      metadata: {
        timestamp: Date.now(),
        source: 'image_input'
      }
    };
  }
  
  // 格式化响应
  formatResponse(result) {
    return {
      success: true,
      data: result,
      timestamp: Date.now()
    };
  }
}

// 业务层
class BusinessLayer {
  constructor() {
    this.services = new Map();
    this.registerServices();
  }
  
  registerServices() {
    this.services.set('chat', new ChatService());
    this.services.set('search', new SearchService());
    this.services.set('recommendation', new RecommendationService());
    this.services.set('analysis', new AnalysisService());
  }
  
  async process(input) {
    // 意图识别
    const intent = await this.recognizeIntent(input);
    
    // 路由到对应服务
    const service = this.services.get(intent.service);
    
    if (!service) {
      throw new Error(`未找到服务: ${intent.service}`);
    }
    
    // 执行业务逻辑
    return await service.execute(input, intent);
  }
  
  async recognizeIntent(input) {
    // 意图识别逻辑
    const { content } = input;
    
    // 关键词匹配
    if (content.includes('搜索') || content.includes('找')) {
      return { service: 'search', confidence: 0.8 };
    }
    
    if (content.includes('推荐') || content.includes('建议')) {
      return { service: 'recommendation', confidence: 0.8 };
    }
    
    if (content.includes('分析') || content.includes('统计')) {
      return { service: 'analysis', confidence: 0.8 };
    }
    
    // 默认为聊天
    return { service: 'chat', confidence: 0.6 };
  }
}

// AI层
class AILayer {
  constructor() {
    this.models = new Map();
    this.cache = new Map();
    this.registerModels();
  }
  
  registerModels() {
    this.models.set('chat', new ChatModel());
    this.models.set('embedding', new EmbeddingModel());
    this.models.set('image', new ImageModel());
    this.models.set('speech', new SpeechModel());
  }
  
  async process(input) {
    const { type, content } = input;
    
    // 检查缓存
    const cacheKey = this.generateCacheKey(input);
    if (this.cache.has(cacheKey)) {
      return this.cache.get(cacheKey);
    }
    
    // 选择模型
    const model = this.selectModel(input);
    
    // 执行推理
    const result = await model.predict(input);
    
    // 缓存结果
    this.cache.set(cacheKey, result);
    
    return result;
  }
  
  selectModel(input) {
    // 根据输入类型选择模型
    if (input.image) {
      return this.models.get('image');
    }
    
    if (input.audio) {
      return this.models.get('speech');
    }
    
    return this.models.get('chat');
  }
  
  generateCacheKey(input) {
    return JSON.stringify({
      type: input.type,
      content: input.content
    });
  }
}

// 数据层
class DataLayer {
  constructor() {
    this.storages = {
      cache: new CacheStorage(),
      persistent: new PersistentStorage(),
      vector: new VectorStorage()
    };
  }
  
  async save(data) {
    // 保存到不同存储
    await Promise.all([
      this.storages.cache.set(data.id, data),
      this.storages.persistent.save(data),
      this.storages.vector.index(data)
    ]);
  }
  
  async get(id) {
    // 先查缓存
    let data = await this.storages.cache.get(id);
    
    if (!data) {
      // 查持久化存储
      data = await this.storages.persistent.get(id);
      
      if (data) {
        // 写入缓存
        await this.storages.cache.set(id, data);
      }
    }
    
    return data;
  }
  
  async search(query) {
    // 向量检索
    return await this.storages.vector.search(query);
  }
}
```

```javascript
// 2. 状态管理实现
class AIStateManager {
  constructor() {
    this.state = this.createInitialState();
    this.listeners = new Map();
    this.middleware = [];
    this.history = [];
  }
  
  createInitialState() {
    return JSON.parse(JSON.stringify(AIAppState));
  }
  
  // 获取状态
  getState(path) {
    return path.split('.').reduce((obj, key) => obj?.[key], this.state);
  }
  
  // 设置状态
  setState(path, value) {
    const keys = path.split('.');
    const lastKey = keys.pop();
    const target = keys.reduce((obj, key) => obj[key], this.state);
    
    const oldValue = target[lastKey];
    target[lastKey] = value;
    
    // 记录历史
    this.history.push({
      path,
      oldValue,
      newValue: value,
      timestamp: Date.now()
    });
    
    // 执行中间件
    this.executeMiddleware(path, value, oldValue);
    
    // 通知监听器
    this.notify(path, value, oldValue);
    
    return this.state;
  }
  
  // 订阅状态变化
  subscribe(path, callback) {
    if (!this.listeners.has(path)) {
      this.listeners.set(path, []);
    }
    
    this.listeners.get(path).push(callback);
    
    // 返回取消订阅函数
    return () => {
      const callbacks = this.listeners.get(path);
      const index = callbacks.indexOf(callback);
      if (index > -1) {
        callbacks.splice(index, 1);
      }
    };
  }
  
  // 通知监听器
  notify(path, newValue, oldValue) {
    const callbacks = this.listeners.get(path) || [];
    callbacks.forEach(callback => callback(newValue, oldValue));
    
    // 通知父路径
    const parentPath = path.split('.').slice(0, -1).join('.');
    if (parentPath) {
      this.notify(parentPath, this.getState(parentPath));
    }
  }
  
  // 添加中间件
  use(middleware) {
    this.middleware.push(middleware);
  }
  
  // 执行中间件
  executeMiddleware(path, newValue, oldValue) {
    this.middleware.forEach(middleware => {
      middleware(path, newValue, oldValue, this.state);
    });
  }
  
  // 批量更新
  batchUpdate(updates) {
    const changes = [];
    
    for (const [path, value] of Object.entries(updates)) {
      const oldValue = this.getState(path);
      changes.push({ path, oldValue, newValue: value });
      
      const keys = path.split('.');
      const lastKey = keys.pop();
      const target = keys.reduce((obj, key) => obj[key], this.state);
      target[lastKey] = value;
    }
    
    // 统一通知
    changes.forEach(({ path, oldValue, newValue }) => {
      this.notify(path, newValue, oldValue);
    });
  }
  
  // 重置状态
  reset() {
    this.state = this.createInitialState();
    this.history = [];
    this.notify('', this.state);
  }
}

// 3. React集成
function useStateManager(path) {
  const stateManager = useContext(StateContext);
  const [value, setValue] = useState(stateManager.getState(path));
  
  useEffect(() => {
    const unsubscribe = stateManager.subscribe(path, (newValue) => {
      setValue(newValue);
    });
    
    return unsubscribe;
  }, [path]);
  
  const update = useCallback((newValue) => {
    stateManager.setState(path, newValue);
  }, [path]);
  
  return [value, update];
}

// 4. 状态持久化
class StatePersistence {
  constructor(stateManager, storageKey) {
    this.stateManager = stateManager;
    this.storageKey = storageKey;
    
    // 加载持久化状态
    this.loadState();
    
    // 监听状态变化
    this.stateManager.subscribe('', this.saveState.bind(this));
  }
  
  loadState() {
    try {
      const saved = localStorage.getItem(this.storageKey);
      if (saved) {
        const parsed = JSON.parse(saved);
        this.stateManager.batchUpdate(parsed);
      }
    } catch (error) {
      console.error('加载状态失败:', error);
    }
  }
  
  saveState() {
    try {
      const state = this.stateManager.state;
      localStorage.setItem(this.storageKey, JSON.stringify(state));
    } catch (error) {
      console.error('保存状态失败:', error);
    }
  }
  
  clear() {
    localStorage.removeItem(this.storageKey);
    this.stateManager.reset();
  }
}
```

## 3. 错误处理与容错

**答案：**
**AI应用错误处理：**

```javascript
class AIErrorHandler {
  constructor() {
    this.errorTypes = {
      NETWORK: 'network',
      MODEL: 'model',
      AUTH: 'auth',
      RATE_LIMIT: 'rate_limit',
      VALIDATION: 'validation',
      UNKNOWN: 'unknown'
    };
    
    this.retryStrategies = new Map();
    this.setupRetryStrategies();
  }
  
  setupRetryStrategies() {
    // 网络错误：指数退避重试
    this.retryStrategies.set(this.errorTypes.NETWORK, {
      maxRetries: 3,
      baseDelay: 1000,
      maxDelay: 10000,
      backoffMultiplier: 2
    });
    
    // 模型错误：立即重试
    this.retryStrategies.set(this.errorTypes.MODEL, {
      maxRetries: 2,
      baseDelay: 0,
      maxDelay: 0,
      backoffMultiplier: 1
    });
    
    // 限流错误：固定延迟重试
    this.retryStrategies.set(this.errorTypes.RATE_LIMIT, {
      maxRetries: 5,
      baseDelay: 5000,
      maxDelay: 30000,
      backoffMultiplier: 1
    });
  }
  
  // 处理错误
  async handleError(error, context = {}) {
    const errorType = this.classifyError(error);
    const strategy = this.retryStrategies.get(errorType);
    
    // 记录错误
    this.logError(error, errorType, context);
    
    // 检查是否需要重试
    if (strategy && context.retryCount < strategy.maxRetries) {
      return await this.retryWithStrategy(error, context, strategy);
    }
    
    // 降级处理
    return await this.fallback(error, errorType, context);
  }
  
  // 错误分类
  classifyError(error) {
    if (error.name === 'NetworkError' || error.message.includes('fetch')) {
      return this.errorTypes.NETWORK;
    }
    
    if (error.message.includes('rate limit') || error.status === 429) {
      return this.errorTypes.RATE_LIMIT;
    }
    
    if (error.status === 401 || error.status === 403) {
      return this.errorTypes.AUTH;
    }
    
    if (error.message.includes('model') || error.message.includes('inference')) {
      return this.errorTypes.MODEL;
    }
    
    if (error.name === 'ValidationError') {
      return this.errorTypes.VALIDATION;
    }
    
    return this.errorTypes.UNKNOWN;
  }
  
  // 重试策略
  async retryWithStrategy(error, context, strategy) {
    const { retryCount = 0 } = context;
    
    // 计算延迟
    const delay = Math.min(
      strategy.baseDelay * Math.pow(strategy.backoffMultiplier, retryCount),
      strategy.maxDelay
    );
    
    // 等待
    await this.sleep(delay);
    
    // 重试
    return { shouldRetry: true, retryCount: retryCount + 1 };
  }
  
  // 降级处理
  async fallback(error, errorType, context) {
    switch (errorType) {
      case this.errorTypes.NETWORK:
        return this.networkFallback(error, context);
      
      case this.errorTypes.MODEL:
        return this.modelFallback(error, context);
      
      case this.errorTypes.AUTH:
        return this.authFallback(error, context);
      
      case this.errorTypes.RATE_LIMIT:
        return this.rateLimitFallback(error, context);
      
      default:
        return this.defaultFallback(error, context);
    }
  }
  
  networkFallback(error, context) {
    return {
      success: false,
      error: '网络连接失败，请检查网络后重试',
      fallbackType: 'offline',
      retryable: true
    };
  }
  
  modelFallback(error, context) {
    return {
      success: false,
      error: 'AI模型暂时不可用，请稍后重试',
      fallbackType: 'model_unavailable',
      retryable: true,
      alternativeModel: this.getAlternativeModel(context.model)
    };
  }
  
  authFallback(error, context) {
    return {
      success: false,
      error: '认证失败，请重新登录',
      fallbackType: 'auth_required',
      retryable: false,
      redirectTo: '/login'
    };
  }
  
  rateLimitFallback(error, context) {
    return {
      success: false,
      error: '请求过于频繁，请稍后重试',
      fallbackType: 'rate_limited',
      retryable: true,
      retryAfter: this.getRetryAfter(error)
    };
  }
  
  defaultFallback(error, context) {
    return {
      success: false,
      error: '发生未知错误，请联系客服',
      fallbackType: 'unknown',
      retryable: false
    };
  }
  
  getAlternativeModel(currentModel) {
    const alternatives = {
      'gpt-4': 'gpt-3.5-turbo',
      'gpt-3.5-turbo': 'claude-2',
      'claude-2': 'gpt-3.5-turbo'
    };
    
    return alternatives[currentModel];
  }
  
  getRetryAfter(error) {
    const retryAfter = error.headers?.['retry-after'];
    return retryAfter ? parseInt(retryAfter) * 1000 : 60000;
  }
  
  logError(error, errorType, context) {
    console.error('AI应用错误:', {
      type: errorType,
      message: error.message,
      stack: error.stack,
      context,
      timestamp: Date.now()
    });
    
    // 发送到监控系统
    this.reportError(error, errorType, context);
  }
  
  async reportError(error, errorType, context) {
    try {
      await fetch('/api/errors', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          type: errorType,
          message: error.message,
          stack: error.stack,
          context,
          url: window.location.href,
          userAgent: navigator.userAgent,
          timestamp: Date.now()
        })
      });
    } catch (e) {
      console.error('报告错误失败:', e);
    }
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

## 4. 性能监控与优化

```javascript
class AIPerformanceMonitor {
  constructor() {
    this.metrics = {
      apiCalls: [],
      modelInference: [],
      rendering: [],
      userInteraction: []
    };
    
    this.thresholds = {
      apiCall: 2000, // 2秒
      modelInference: 5000, // 5秒
      rendering: 16, // 16ms (60fps)
      userInteraction: 100 // 100ms
    };
  }
  
  // 监控API调用
  async monitorAPICall(apiCall, metadata = {}) {
    const startTime = performance.now();
    const startMemory = this.getMemoryUsage();
    
    try {
      const result = await apiCall();
      
      const endTime = performance.now();
      const endMemory = this.getMemoryUsage();
      
      const metric = {
        type: 'api_call',
        duration: endTime - startTime,
        memoryDelta: endMemory - startMemory,
        success: true,
        metadata,
        timestamp: Date.now()
      };
      
      this.recordMetric('apiCalls', metric);
      this.checkThreshold(metric);
      
      return result;
      
    } catch (error) {
      const metric = {
        type: 'api_call',
        duration: performance.now() - startTime,
        success: false,
        error: error.message,
        metadata,
        timestamp: Date.now()
      };
      
      this.recordMetric('apiCalls', metric);
      throw error;
    }
  }
  
  // 监控模型推理
  async monitorModelInference(inferenceCall, modelInfo = {}) {
    const startTime = performance.now();
    
    const result = await inferenceCall();
    
    const duration = performance.now() - startTime;
    
    const metric = {
      type: 'model_inference',
      model: modelInfo.name,
      inputTokens: modelInfo.inputTokens,
      outputTokens: modelInfo.outputTokens,
      duration,
      tokensPerSecond: (modelInfo.outputTokens / duration) * 1000,
      timestamp: Date.now()
    };
    
    this.recordMetric('modelInference', metric);
    this.checkThreshold(metric);
    
    return result;
  }
  
  // 监控渲染性能
  monitorRendering(componentName, renderFunction) {
    const startTime = performance.now();
    
    const result = renderFunction();
    
    const duration = performance.now() - startTime;
    
    const metric = {
      type: 'rendering',
      component: componentName,
      duration,
      timestamp: Date.now()
    };
    
    this.recordMetric('rendering', metric);
    this.checkThreshold(metric);
    
    return result;
  }
  
  // 监控用户交互
  monitorInteraction(interactionName, interactionFunction) {
    const startTime = performance.now();
    
    const result = interactionFunction();
    
    const duration = performance.now() - startTime;
    
    const metric = {
      type: 'user_interaction',
      interaction: interactionName,
      duration,
      timestamp: Date.now()
    };
    
    this.recordMetric('userInteraction', metric);
    this.checkThreshold(metric);
    
    return result;
  }
  
  // 记录指标
  recordMetric(category, metric) {
    this.metrics[category].push(metric);
    
    // 限制历史记录
    if (this.metrics[category].length > 1000) {
      this.metrics[category] = this.metrics[category].slice(-500);
    }
  }
  
  // 检查阈值
  checkThreshold(metric) {
    let threshold;
    
    switch (metric.type) {
      case 'api_call':
        threshold = this.thresholds.apiCall;
        break;
      case 'model_inference':
        threshold = this.thresholds.modelInference;
        break;
      case 'rendering':
        threshold = this.thresholds.rendering;
        break;
      case 'user_interaction':
        threshold = this.thresholds.userInteraction;
        break;
    }
    
    if (metric.duration > threshold) {
      this.reportSlowPerformance(metric, threshold);
    }
  }
  
  // 报告性能问题
  reportSlowPerformance(metric, threshold) {
    console.warn('性能警告:', {
      type: metric.type,
      duration: metric.duration,
      threshold,
      exceedBy: metric.duration - threshold
    });
    
    // 发送到监控系统
    this.sendToMonitoring(metric, threshold);
  }
  
  // 获取内存使用情况
  getMemoryUsage() {
    if (performance.memory) {
      return performance.memory.usedJSHeapSize;
    }
    return 0;
  }
  
  // 生成性能报告
  generateReport() {
    const report = {
      timestamp: Date.now(),
      summary: {
        totalAPICalls: this.metrics.apiCalls.length,
        averageAPICallDuration: this.calculateAverage(this.metrics.apiCalls),
        totalModelInferences: this.metrics.modelInference.length,
        averageInferenceDuration: this.calculateAverage(this.metrics.modelInference),
        slowRenders: this.metrics.rendering.filter(m => m.duration > this.thresholds.rendering).length
      },
      details: this.metrics
    };
    
    return report;
  }
```

```javascript
    if (input.length > policy.maxLength) {
      errors.push(`输入长度超过限制: ${input.length} > ${policy.maxLength}`);
    }

    if (!policy.allowedCharacters.test(input)) {

    for (const pattern of policy.blockedPatterns) {

        errors.push('输入包含潜在的XSS攻击');
        break;
      }
    }
    
    return {
      valid: errors.length === 0,
      errors,
      sanitized: this.sanitizeInput(input)
    };
  }
  
  // 内容过滤
  filterContent(content) {
    const policy = this.policies.get('content_filter');
    const warnings = [];
    
    // 检查敏感话题
```
