## 3. 谈谈你对Prompt Engineering的理解，前端如何通过工程化手段提升Prompt的质量和安全性？

**答案：**
**Prompt Engineering核心理解：**
- 不仅仅是写提示词，而是系统化的工程实践

**前端工程化手段：**

1. **Prompt模板系统**：
   ```javascript
   class PromptTemplate {
     constructor(template, variables = []) {
       this.template = template;
       this.variables = variables;
     }
     
     render(context) {
       let result = this.template;
       for (const [key, value] of Object.entries(context)) {
         result = result.replace(new RegExp(`{{${key}}}`, 'g'), value);
       }
       return result;
     }
     
     validate(context) {
       const missing = this.variables.filter(v => !(v in context));
       if (missing.length > 0) {
         throw new Error(`缺少必要变量: ${missing.join(', ')}`);
       }
     }
   }
   
   // 使用示例
   const qaPrompt = new PromptTemplate(
     `基于以下上下文回答问题：
   上下文：{{context}}
   问题：{{question}}
   请用中文回答。`,
     ['context', 'question']
   );
   ```

2. **Prompt版本管理**：
   ```javascript
   class PromptRegistry {
     constructor() {
       this.prompts = new Map();
       this.versions = new Map();
     }
     
     register(name, prompt, version = '1.0.0') {
       if (!this.prompts.has(name)) {
         this.prompts.set(name, []);
       }
       this.prompts.get(name).push({
         version,
         prompt,
         createdAt: new Date()
       });
     }
     
     get(name, version) {
       const versions = this.prompts.get(name);
       if (!versions) return null;
       return versions.find(v => v.version === version);
     }
     
     getLatest(name) {
       const versions = this.prompts.get(name);
       if (!versions) return null;
       return versions[versions.length - 1];
     }
   }
   ```

## 7. 你们项目中是如何处理AI生成内容的持久化和多端同步的？

   class LocalStorageManager {
     constructor(key) {
       this.key = key;
     }
     
     save(data) {
       const serialized = JSON.stringify({
         data,
         timestamp: Date.now(),
         version: '1.0'
       });
       localStorage.setItem(this.key, serialized);
     }
     
     load() {
       const item = localStorage.getItem(this.key);
       if (!item) return null;
       return JSON.parse(item).data;
     }
     
     clear() {
       localStorage.removeItem(this.key);
     }

2. **IndexedDB存储**：
   ```javascript
   class IndexedDBManager {
     constructor(dbName, storeName) {
       this.dbName = dbName;
       this.storeName = storeName;
     }
     
     async open() {
       return new Promise((resolve, reject) => {
         const request = indexedDB.open(this.dbName, 1);
         request.onupgradeneeded = (event) => {
           const db = event.target.result;
           if (!db.objectStoreNames.contains(this.storeName)) {
             db.createObjectStore(this.storeName, { keyPath: 'id' });
           }
         };
         request.onsuccess = (event) => resolve(event.target.result);
         request.onerror = (event) => reject(event.target.error);
       });
     }
     
     async save(data) {
       const db = await this.open();
       return new Promise((resolve, reject) => {
         const transaction = db.transaction([this.storeName], 'readwrite');
         const store = transaction.objectStore(this.storeName);
         const request = store.put(data);
         request.onsuccess = () => resolve();
         request.onerror = () => reject(request.error);
       });
     }
   }
   ```

**多端同步方案：**

1. **基于WebSocket的实时同步**：
   ```javascript
   class SyncManager {
     constructor(wsUrl) {
       this.ws = new WebSocket(wsUrl);
       this.listeners = new Map();
       
       this.ws.onmessage = (event) => {
         const data = JSON.parse(event.data);
         this.notifyListeners(data.type, data.payload);
       };
     }
     
     subscribe(type, callback) {
       if (!this.listeners.has(type)) {
         this.listeners.set(type, []);
       }
       this.listeners.get(type).push(callback);
     }
     
     notifyListeners(type, payload) {
       const callbacks = this.listeners.get(type) || [];
       callbacks.forEach(cb => cb(payload));
     }
     
     send(type, payload) {
       this.ws.send(JSON.stringify({ type, payload }));
     }
   }
   ```

   class ConflictResolver {
     // 时间戳优先
     resolveByTimestamp(local, remote) {
       return local.timestamp > remote.timestamp ? local : remote;
     }
     
     // 版本号优先
     resolveByVersion(local, remote) {
       return local.version > remote.version ? local : remote;
     }
     
     // 合并策略
     merge(local, remote) {
       return {
         ...remote,
         ...local,
         metadata: {
           ...remote.metadata,
           ...local.metadata,
           lastModified: Math.max(local.metadata.lastModified, remote.metadata.lastModified)
         }

3. **流式数据处理**：
   ```javascript
   // worker.js
   self.onmessage = function(e) {
     const { action, chunk } = e.data;
     
     if (action === 'process') {
       // 处理流式数据
       const processed = processChunk(chunk);
       self.postMessage({ action: 'processed', data: processed });
     }
   };
   
   function processChunk(chunk) {
     // 解析SSE数据
     const lines = chunk.split('\n');
     const events = [];
     
     for (const line of lines) {
       if (line.startsWith('data: ')) {
         const data = line.slice(6);
         if (data === '[DONE]') {
           events.push({ type: 'done' });
         } else {
           try {
             events.push({ type: 'data', data: JSON.parse(data) });
           } catch (e) {
             events.push({ type: 'error', error: e.message });
           }
         }
       }
     }
     
     return events;
   }
   ```

   class WasmChunkLoader {
     constructor(chunks) {
       this.chunks = chunks;
       this.loaded = new Map();
     }
     
     async loadChunk(chunkId) {
       if (this.loaded.has(chunkId)) {
         return this.loaded.get(chunkId);
       }
       
       const response = await fetch(`/wasm/${chunkId}.wasm`);
       const buffer = await response.arrayBuffer();
       const module = await WebAssembly.compile(buffer);
       
       this.loaded.set(chunkId, module);
       return module;
     }
     
     async loadAll() {
       const promises = this.chunks.map(chunk => this.loadChunk(chunk.id));
       return Promise.all(promises);
     }
   }

   class WasmMemoryManager {
     constructor(initialSize = 1024 * 1024) { // 1MB
       this.memory = new WebAssembly.Memory({ 
         initial: initialSize / 65536, // 页大小64KB
         maximum: 1024 * 1024 * 1024 / 65536 // 最大1GB
       });
       this.allocated = new Map();
     }
     
     allocate(size) {
       // 简单的内存分配器
       const pages = Math.ceil(size / 65536);
       const currentPages = this.memory.buffer.byteLength / 65536;
       
       if (currentPages < pages) {
         this.memory.grow(pages - currentPages);
       }
       
       return this.memory.buffer;
     }
     
     free(ptr) {
       // 标记内存为可回收
       this.allocated.delete(ptr);
     }
   }

   class PermissionManager {
     constructor() {
       this.permissions = new Map();
     }
     
     grant(userId, permission) {
       if (!this.permissions.has(userId)) {
         this.permissions.set(userId, new Set());
       }
       this.permissions.get(userId).add(permission);
     }
     
     check(userId, permission) {
       const userPermissions = this.permissions.get(userId);
       return userPermissions ? userPermissions.has(permission) : false;
     }
     
     revoke(userId, permission) {
       const userPermissions = this.permissions.get(userId);
       if (userPermissions) {
         userPermissions.delete(permission);
       }
     }
   }
