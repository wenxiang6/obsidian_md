## 1. SSE和Streamable HTTP的区别？

**答案：**
**SSE（Server-Sent Events）：**
- 基于HTTP/1.1的长连接
- 单向通信：服务器→客户端
- 文本格式（UTF-8）
- 自动重连机制
- 简单易用，浏览器原生支持

**Streamable HTTP：**
- 基于HTTP/2或HTTP/3
- 支持双向流式传输
- 二进制数据支持
- 更好的多路复用
- 需要polyfill或库支持

**核心区别：**

| 特性 | SSE | Streamable HTTP |
|------|-----|-----------------|
| 通信方向 | 单向（服务器→客户端） | 双向 |
| 数据格式 | 文本 | 二进制/文本 |
| 协议版本 | HTTP/1.1 | HTTP/2+ |
| 多路复用 | 不支持 | 支持 |
| 浏览器支持 | 原生支持 | 需要polyfill |
| 复杂度 | 简单 | 复杂 |

**前端实现示例：**

```javascript
// SSE实现
class SSEClient {
  constructor(url) {
    this.url = url;
    this.eventSource = null;
  }
  
  connect() {
    this.eventSource = new EventSource(this.url);
    
    this.eventSource.onmessage = (event) => {
      console.log('收到消息:', event.data);
    };
    
    this.eventSource.onerror = (error) => {
      console.error('SSE错误:', error);
      this.reconnect();
    };
  }
  
  reconnect() {
    setTimeout(() => this.connect(), 3000);
  }
  
  close() {
    if (this.eventSource) {
      this.eventSource.close();
    }
  }
}

// Streamable HTTP实现（使用fetch + ReadableStream）
class StreamableHTTPClient {
  constructor(url) {
    this.url = url;
  }
  
  async stream(body) {
    const response = await fetch(this.url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body)
    });
    
    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      
      const text = decoder.decode(value, { stream: true });
      console.log('收到数据:', text);
    }
  }
}
```

```javascript
// 传输适配器接口
class TransportAdapter {
  constructor(config) {
    this.config = config;
  }
  
  async connect() { throw new Error('未实现'); }
  async send(message) { throw new Error('未实现'); }
  async close() { throw new Error('未实现'); }
  onMessage(callback) { this.messageCallback = callback; }
}

// stdio适配器（用于本地进程通信）
class StdioAdapter extends TransportAdapter {
  constructor(config) {
    super(config);
    this.process = null;
  }
  
  async connect() {
    const { spawn } = require('child_process');
    this.process = spawn(this.config.command, this.config.args);
    
    this.process.stdout.on('data', (data) => {
      if (this.messageCallback) {
        this.messageCallback(data.toString());
      }
    });
    
    this.process.stderr.on('data', (data) => {
      console.error('stderr:', data.toString());
    });
  }
  
  async send(message) {
    this.process.stdin.write(JSON.stringify(message) + '\n');
  }
  
  async close() {
    this.process.kill();
  }
}

// HTTP适配器
class HTTPAdapter extends TransportAdapter {
  constructor(config) {
    super(config);
    this.polling = false;
  }
  
  async connect() {
    // HTTP通常是无状态的，不需要持久连接
    // 可以开始轮询
    this.startPolling();
  }
  
  async send(message) {
    const response = await fetch(this.config.url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(message)
    });
    
    return response.json();
  }
  
  async startPolling() {
    this.polling = true;
    while (this.polling) {
      try {
        const response = await fetch(this.config.pollUrl);
        const data = await response.json();
        if (this.messageCallback) {
          this.messageCallback(data);
        }
      } catch (error) {
        console.error('轮询错误:', error);
      }
      await new Promise(resolve => setTimeout(resolve, 1000));
    }
  }
  
  async close() {
    this.polling = false;
  }
}

// SSE适配器
class SSEAdapter extends TransportAdapter {
  constructor(config) {
    super(config);
    this.eventSource = null;
  }
  
  async connect() {
    this.eventSource = new EventSource(this.config.url);
    
    this.eventSource.onmessage = (event) => {
      if (this.messageCallback) {
        this.messageCallback(JSON.parse(event.data));
      }
    };
    
    this.eventSource.onerror = (error) => {
      console.error('SSE错误:', error);
    };
  }
  
  async send(message) {
    // SSE是单向的，需要通过HTTP POST发送
    await fetch(this.config.sendUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(message)
    });
  }
  
  async close() {
    if (this.eventSource) {
      this.eventSource.close();
    }
  }
}

// Streamable HTTP适配器
class StreamableHTTPAdapter extends TransportAdapter {
  constructor(config) {
    super(config);
    this.reader = null;
  }
  
  async connect() {
    // 建立流式连接
    const response = await fetch(this.config.url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ action: 'connect' })
    });
    
    this.reader = response.body.getReader();
    this.readStream();
  }
  
  async readStream() {
    const decoder = new TextDecoder();
    
    while (true) {
      const { done, value } = await this.reader.read();
      if (done) break;
      
      const text = decoder.decode(value, { stream: true });
      if (this.messageCallback) {
        this.messageCallback(JSON.parse(text));
      }
    }
  }
  
  async send(message) {
    const response = await fetch(this.config.url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(message)
    });
    
    return response.json();
  }
  
  async close() {
    if (this.reader) {
      await this.reader.cancel();
    }
  }
}

// 工厂函数
function createTransport(type, config) {
  switch (type) {
    case 'stdio': return new StdioAdapter(config);
    case 'http': return new HTTPAdapter(config);
    case 'sse': return new SSEAdapter(config);
    case 'streamable-http': return new StreamableHTTPAdapter(config);
    default: throw new Error(`不支持的传输类型: ${type}`);
  }
}
```

## 3. 如何处理MCP版本兼容问题？

**答案：**
**版本兼容策略：**

1. **版本协商**：
   ```javascript
   class MCPVersionManager {
     constructor() {
       this.supportedVersions = ['1.0', '1.1', '2.0'];
       this.currentVersion = null;
     }
     
     async negotiateVersion(serverCapabilities) {
       // 获取服务器支持的版本
       const serverVersions = serverCapabilities.versions || [];
       
       // 找到共同支持的版本
       const commonVersions = this.supportedVersions.filter(v => 
         serverVersions.includes(v)
       );
       
       if (commonVersions.length === 0) {
         throw new Error('没有共同支持的版本');
       }
       
       // 选择最新的共同版本
       this.currentVersion = commonVersions.sort().reverse()[0];
       return this.currentVersion;
     }
     
     getCapabilities() {
       switch (this.currentVersion) {
         case '1.0':
           return {
             tools: true,
             resources: false,
             prompts: false
           };
         case '1.1':
           return {
             tools: true,
             resources: true,
             prompts: false
           };
         case '2.0':
           return {
             tools: true,
             resources: true,
             prompts: true
           };
         default:
           return {};
       }
     }
   }
   ```

2. **特性检测**：
   ```javascript
   class FeatureDetector {
     constructor(versionManager) {
       this.versionManager = versionManager;
     }
     
     hasFeature(feature) {
       const capabilities = this.versionManager.getCapabilities();
       return capabilities[feature] === true;
     }
     
     requireFeature(feature) {
       if (!this.hasFeature(feature)) {
         throw new Error(`当前版本不支持特性: ${feature}`);
       }
     }
   }
   ```

3. **降级处理**：
   ```javascript
   class GracefulDegradation {
     constructor(versionManager) {
       this.versionManager = versionManager;
     }
     
     async callWithFallback(primaryCall, fallbackCall) {
       try {
         if (this.versionManager.hasFeature('advanced')) {
           return await primaryCall();
         } else {
           return await fallbackCall();
         }
       } catch (error) {
         console.warn('高级特性调用失败，降级处理:', error);
         return await fallbackCall();
       }
     }
   }
   ```

4. **版本迁移**：
   ```javascript
   class VersionMigrator {
     migrate(data, fromVersion, toVersion) {
       const migrations = {
         '1.0->1.1': this.migrate1_0To1_1,
         '1.1->2.0': this.migrate1_1To2_0,
         '1.0->2.0': this.migrate1_0To2_0

       const key = `${fromVersion}->${toVersion}`;
       const migration = migrations[key];

     }
     
     migrate1_0To1_1(data) {
       // 1.0到1.1的迁移逻辑
       return {

         version: '1.1',

       };
     }
     
     migrate1_1To2_0(data) {
       // 1.1到2.0的迁移逻辑

         ...data,
         version: '2.0',
         prompts: [] // 新增提示词字段
       };
     }
     
     migrate1_0To2_0(data) {
       // 1.0到2.0的迁移逻辑
       const intermediate = this.migrate1_0To1_1(data);
       return this.migrate1_1To2_0(intermediate);
   ```

## 4. ChatGPT这种对话Agent的会话表和消息表字段怎么设计？

**答案：**
**数据库设计：**

1. **会话表（conversations）**：
   ```sql
   CREATE TABLE conversations (
     id VARCHAR(36) PRIMARY KEY,
     user_id VARCHAR(36) NOT NULL,
     title VARCHAR(255),
     model VARCHAR(50) NOT NULL,
     system_prompt TEXT,
     temperature DECIMAL(3,2) DEFAULT 0.7,
     max_tokens INT DEFAULT 4096,
     status ENUM('active', 'archived', 'deleted') DEFAULT 'active',
     metadata JSON, -- 存储额外配置
     created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
     updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
     last_message_at TIMESTAMP,
     
     INDEX idx_user_id (user_id),
     INDEX idx_status (status),
     INDEX idx_last_message (last_message_at)
   );
   ```

2. **消息表（messages）**：
   ```sql
   CREATE TABLE messages (
     id VARCHAR(36) PRIMARY KEY,
     conversation_id VARCHAR(36) NOT NULL,
     role ENUM('system', 'user', 'assistant', 'tool') NOT NULL,
     content TEXT,
     token_count INT,
     model VARCHAR(50),
     finish_reason VARCHAR(50),
     tool_calls JSON, -- 工具调用信息
     tool_call_id VARCHAR(36), -- 工具调用ID
     parent_message_id VARCHAR(36), -- 用于分支对话
     metadata JSON,
     created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
     
     FOREIGN KEY (conversation_id) REFERENCES conversations(id) ON DELETE CASCADE,
     INDEX idx_conversation_id (conversation_id),
     INDEX idx_role (role),
     INDEX idx_created_at (created_at)
   );
   ```

3. **工具调用表（tool_calls）**：
   ```sql
   CREATE TABLE tool_calls (
     id VARCHAR(36) PRIMARY KEY,
     message_id VARCHAR(36) NOT NULL,
     tool_name VARCHAR(100) NOT NULL,
     arguments JSON,
     result JSON,
     status ENUM('pending', 'running', 'completed', 'failed') DEFAULT 'pending',
     error_message TEXT,
     execution_time_ms INT,
     created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
     completed_at TIMESTAMP,
     
     FOREIGN KEY (message_id) REFERENCES messages(id) ON DELETE CASCADE,
     INDEX idx_message_id (message_id),
     INDEX idx_tool_name (tool_name),
     INDEX idx_status (status)
   );
   ```

4. **会话统计表（conversation_stats）**：
   ```sql
   CREATE TABLE conversation_stats (
     conversation_id VARCHAR(36) PRIMARY KEY,
     total_messages INT DEFAULT 0,
     total_tokens INT DEFAULT 0,
     total_tool_calls INT DEFAULT 0,
     avg_response_time_ms INT,
     user_satisfaction_score DECIMAL(3,2),
     last_activity_at TIMESTAMP,
     
     FOREIGN KEY (conversation_id) REFERENCES conversations(id) ON DELETE CASCADE
   );
   ```

**字段说明：**
- **conversations**: 存储会话基本信息，包括模型配置、状态等
- **messages**: 存储每条消息，支持分支对话（通过parent_message_id）
- **tool_calls**: 存储工具调用详情，支持异步执行和状态追踪
- **conversation_stats**: 存储会话统计信息，便于分析和优化

## 5. MCP的客户端和服务端交互完整流程？

**答案：**
**完整交互流程：**

```javascript
// 1. 客户端初始化
class MCPClient {
  constructor(transport) {
    this.transport = transport;
    this.serverCapabilities = null;
    this.requestId = 0;
    this.pendingRequests = new Map();
  }
  
  async initialize() {
    // 发送初始化请求
    const response = await this.request('initialize', {
      protocolVersion: '2.0',
      capabilities: {
        roots: { listChanged: true }
      },
      clientInfo: {
        name: 'MyClient',
        version: '1.0.0'
      }
    });
    
    this.serverCapabilities = response.capabilities;
    
    // 发送初始化完成通知
    await this.notify('initialized', {});
    
    return response;
  }
  
  async request(method, params) {
    const id = ++this.requestId;
    const request = {
      jsonrpc: '2.0',
      id,
      method,
      params
    };
    
    return new Promise((resolve, reject) => {
      this.pendingRequests.set(id, { resolve, reject });
      this.transport.send(request);
    });
  }
  
  async notify(method, params) {
    const notification = {
      jsonrpc: '2.0',
      method,
      params
    };
    
    await this.transport.send(notification);
  }
  
  handleResponse(response) {
    const pending = this.pendingRequests.get(response.id);
    if (pending) {
      if (response.error) {
        pending.reject(new Error(response.error.message));
      } else {
        pending.resolve(response.result);
      }
      this.pendingRequests.delete(response.id);
    }
  }
}

// 2. 服务端实现
class MCPServer {
  constructor() {
    this.tools = new Map();
    this.resources = new Map();
    this.prompts = new Map();
  }
  
  async handleRequest(request) {
    switch (request.method) {
      case 'initialize':
        return this.handleInitialize(request.params);
      case 'tools/list':
        return this.handleListTools();
      case 'tools/call':
        return this.handleCallTool(request.params);
      case 'resources/list':
        return this.handleListResources();
      case 'resources/read':
        return this.handleReadResource(request.params);
      case 'prompts/list':
        return this.handleListPrompts();
      case 'prompts/get':
        return this.handleGetPrompt(request.params);
      default:
        throw new Error(`未知方法: ${request.method}`);
    }
  }
  
  handleInitialize(params) {
    return {
      protocolVersion: '2.0',
      capabilities: {
        tools: { listChanged: true },
        resources: { subscribe: true, listChanged: true },
        prompts: { listChanged: true }
      },
      serverInfo: {
        name: 'MyServer',
        version: '1.0.0'
      }
    };
  }
  
  handleListTools() {
    return {
      tools: Array.from(this.tools.values())
    };
  }
  
  async handleCallTool(params) {
    const { name, arguments: args } = params;
    const tool = this.tools.get(name);
    
    if (!tool) {
      throw new Error(`工具不存在: ${name}`);
    }
    
    try {
      const result = await tool.execute(args);
      return {
        content: [{ type: 'text', text: JSON.stringify(result) }]
      };
    } catch (error) {
      return {
        content: [{ type: 'text', text: `错误: ${error.message}` }],
        isError: true
      };
    }
  }
}

// 3. 完整使用示例

  // 创建传输层

    command: 'node',

  });
  
  // 创建客户端

}
```

```javascript
class MCPClient {
  async listTools() {
    // 1. 检查服务器能力
    if (!this.serverCapabilities?.tools) {
      throw new Error('服务器不支持工具功能');
    }
    
    // 2. 发送请求
    const response = await this.request('tools/list', {});
    
    // 3. 验证响应
    if (!response || !Array.isArray(response.tools)) {
      throw new Error('无效的工具列表响应');
    }
    
    // 4. 缓存工具列表
    this.toolsCache = response.tools;
    
    return response.tools;
  }
  
  // 获取工具详情
  async getToolDetails(toolName) {
    const tools = await this.listTools();
    const tool = tools.find(t => t.name === toolName);
    
    if (!tool) {
      throw new Error(`工具不存在: ${toolName}`);
    }
    
    return tool;
  }
  
  // 搜索工具
  async searchTools(query) {
    const tools = await this.listTools();
    
    return tools.filter(tool => {
      const searchText = `${tool.name} ${tool.description}`.toLowerCase();
      return searchText.includes(query.toLowerCase());
    });
  }

class MCPServer {

    this.toolsListChanged = false;

      name,
      description,
      inputSchema,

    // 通知客户端（如果支持）
    this.notifyToolsChanged();

      description: tool.description,

    return { tools };

      });
    }
  }
```

## 7. 如果MCP有很多工具，比如100个，Agent怎么处理？

**答案：**
**工具管理策略：**

1. **工具分类与索引**：
   ```javascript
   class ToolRegistry {
     constructor() {
       this.tools = new Map();
       this.categories = new Map();
       this.index = new Map(); // 用于快速搜索
     }
     
     registerTool(tool) {
       // 1. 注册工具
       this.tools.set(tool.name, tool);
       
       // 2. 分类
       const category = tool.category || '未分类';
       if (!this.categories.has(category)) {
         this.categories.set(category, []);
       }
       this.categories.get(category).push(tool.name);
       
       // 3. 建立索引
       this.buildIndex(tool);
     }
     
     buildIndex(tool) {
       // 建立关键词索引
       const keywords = this.extractKeywords(tool);
       for (const keyword of keywords) {
         if (!this.index.has(keyword)) {
           this.index.set(keyword, []);
         }
         this.index.get(keyword).push(tool.name);
       }
     }
     
     extractKeywords(tool) {
       const text = `${tool.name} ${tool.description}`.toLowerCase();
       return text.split(/\s+/).filter(word => word.length > 2);
     }
     
     // 按类别获取工具
     getToolsByCategory(category) {
       const toolNames = this.categories.get(category) || [];
       return toolNames.map(name => this.tools.get(name));
     }
     
     // 搜索工具
     searchTools(query) {
       const keywords = query.toLowerCase().split(/\s+/);
       const scores = new Map();
       
       for (const keyword of keywords) {
         const toolNames = this.index.get(keyword) || [];
         for (const name of toolNames) {
           scores.set(name, (scores.get(name) || 0) + 1);
         }
       }
       
       // 按相关性排序
       return Array.from(scores.entries())
         .sort((a, b) => b[1] - a[1])
         .map(([name]) => this.tools.get(name));
     }
   ```

   ```javascript
     constructor(maxSize = 50) {
       this.cache = new Map();
       this.maxSize = maxSize;
       this.usageOrder = [];
     }
     
     get(toolName) {
       if (this.cache.has(toolName)) {
         // 更新使用顺序
         this.updateUsageOrder(toolName);
         return this.cache.get(toolName);
       }
       return null;
     }
     
     set(toolName, tool) {
       // 如果缓存已满，移除最久未使用的
       if (this.cache.size >= this.maxSize) {
         const oldest = this.usageOrder.shift();
         this.cache.delete(oldest);
       }
       
       this.cache.set(toolName, tool);
       this.usageOrder.push(toolName);
     }
     
     updateUsageOrder(toolName) {
       const index = this.usageOrder.indexOf(toolName);
       if (index > -1) {
         this.usageOrder.splice(index, 1);
       }
       this.usageOrder.push(toolName);
     }
     
     // 预加载相关工具
     preload(toolNames) {
       for (const name of toolNames) {
         if (!this.cache.has(name)) {
           // 异步加载
           this.loadTool(name).then(tool => {
             this.set(name, tool);
           });
         }
       }
     }
   }
   ```

```javascript
class LoopDetector {
  constructor(config = {}) {
    this.maxIterations = config.maxIterations || 10;
    this.maxTime = config.maxTime || 300000; // 5分钟
    this.similarityThreshold = config.similarityThreshold || 0.9;
    
    this.history = [];
    this.startTime = null;
  }
  
  // 检查是否应该继续
  shouldContinue(currentState) {
    // 1. 检查迭代次数
    if (this.history.length >= this.maxIterations) {
      console.warn('达到最大迭代次数');
      return false;
    }
    
    // 2. 检查时间限制
    if (Date.now() - this.startTime > this.maxTime) {
      console.warn('达到最大执行时间');
      return false;
    }
    
    // 3. 检查状态重复
    if (this.isStateRepeating(currentState)) {
      console.warn('检测到状态重复');
      return false;
    }
    
    // 4. 检查进展停滞
    if (this.isProgressStalled()) {
      console.warn('检测到进展停滞');
      return false;
    }
    
    return true;
  }
  
  // 记录状态
  recordState(state) {
    this.history.push({
      state,
      timestamp: Date.now()
    });
    
    if (!this.startTime) {
      this.startTime = Date.now();
    }
  }
  
  // 检查状态重复
  isStateRepeating(currentState) {
    if (this.history.length < 3) return false;
    
    // 检查最近N个状态是否相似
    const recentStates = this.history.slice(-3);
    const similarities = recentStates.map(prev => 
      this.calculateSimilarity(prev.state, currentState)
    );
    
    return similarities.every(sim => sim > this.similarityThreshold);
  }
  
  // 检查进展停滞
  isProgressStalled() {
    if (this.history.length < 5) return false;
    
    // 检查最近5个状态是否有实质进展
    const recentStates = this.history.slice(-5);
    const progress = this.calculateProgress(recentStates);
    
    return progress < 0.1; // 进展小于10%
  }
  
  // 计算状态相似度
  calculateSimilarity(state1, state2) {
    // 简单实现：基于字符串相似度
    const str1 = JSON.stringify(state1);
    const str2 = JSON.stringify(state2);
    
    // 使用Levenshtein距离或其他算法
    const maxLen = Math.max(str1.length, str2.length);
    if (maxLen === 0) return 1;
    
    const distance = this.levenshteinDistance(str1, str2);
    return 1 - distance / maxLen;
  }
  
  // 计算进展
  calculateProgress(states) {
    // 简单实现：基于状态变化
    let totalChange = 0;
    
    for (let i = 1; i < states.length; i++) {
      const change = this.calculateStateChange(states[i-1].state, states[i].state);
      totalChange += change;
    }
    
    return totalChange / (states.length - 1);
  }
  
  // Levenshtein距离算法
  levenshteinDistance(str1, str2) {
    const m = str1.length;
    const n = str2.length;
    const dp = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0));
    
    for (let i = 0; i <= m; i++) dp[i][0] = i;
    for (let j = 0; j <= n; j++) dp[0][j] = j;
    
    for (let i = 1; i <= m; i++) {
      for (let j = 1; j <= n; j++) {
        if (str1[i-1] === str2[j-1]) {
          dp[i][j] = dp[i-1][j-1];
        } else {
          dp[i][j] = 1 + Math.min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1]);
        }
      }
    }
    
    return dp[m][n];
  }
}

class SafeAgent {
  constructor(config) {
    this.loopDetector = new LoopDetector(config.loopDetection);
    this.maxRetries = config.maxRetries || 3;
  }
  
  async executeWithLoopDetection(task) {
    let currentState = { task, step: 0 };
    let retryCount = 0;
    
    while (this.loopDetector.shouldContinue(currentState)) {
      try {
        // 记录当前状态
        this.loopDetector.recordState(currentState);
        
        // 执行任务
        const result = await this.executeStep(currentState);
        
        // 更新状态
        currentState = this.updateState(currentState, result);
        
        // 检查是否完成
        if (this.isTaskComplete(currentState)) {
          return result;
        }
        
        // 重置重试计数
        retryCount = 0;
        
      } catch (error) {
        retryCount++;
        
        if (retryCount >= this.maxRetries) {
          throw new Error(`任务执行失败，重试${this.maxRetries}次后放弃: ${error.message}`);
        }
        
        // 等待后重试
        await this.delay(1000 * retryCount);
      }
    }
    
    // 检测到死循环，执行恢复策略
    return await this.recoverFromLoop(currentState);
  }
  
  async recoverFromLoop(state) {
    // 恢复策略
    console.log('执行死循环恢复策略');
    
    // 1. 简化任务
    const simplifiedTask = await this.simplifyTask(state.task);
    
    // 2. 使用备用方法
    const backupResult = await this.executeBackupMethod(simplifiedTask);
    
    // 3. 记录失败原因
    await this.recordFailureReason(state);
    
    return backupResult;
  }
```
