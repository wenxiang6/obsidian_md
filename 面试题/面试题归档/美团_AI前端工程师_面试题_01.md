```javascript
// Agent协作系统设计
class MultiAgentSystem {
  constructor() {
    this.agents = new Map();
    this.taskQueue = [];
    this.results = new Map();
  }
  
  // 注册Agent
  registerAgent(agent) {
    this.agents.set(agent.id, agent);
  }
  
  // 分配任务
  async assignTask(task) {
    const agent = this.selectAgent(task);
    
    if (!agent) {
      throw new Error('没有合适的Agent处理此任务');
    }
    
    const result = await agent.execute(task);
    this.results.set(task.id, result);
    
    return result;
  }
  
  // 选择Agent
  selectAgent(task) {
    const candidates = Array.from(this.agents.values())
      .filter(agent => agent.canHandle(task));
    
    if (candidates.length === 0) return null;
    
    // 基于负载、能力、历史表现选择
    return candidates.reduce((best, current) => {
      const bestScore = this.calculateAgentScore(best, task);
      const currentScore = this.calculateAgentScore(current, task);
      return currentScore > bestScore ? current : best;
    });
  }
  
  calculateAgentScore(agent, task) {
    let score = 0;
    
    // 能力匹配度
    score += agent.getCapabilityMatch(task) * 40;
    
    // 当前负载（负载越低分数越高）
    score += (1 - agent.getLoad()) * 30;
    
    // 历史成功率
    score += agent.getSuccessRate() * 20;
    
    // 响应速度
    score += (1 / agent.getAverageResponseTime()) * 10;
    
    return score;
  }
}

class CodingAssistantOrchestrator {
  constructor() {
    this.agentSystem = new MultiAgentSystem();
    this.workflow = new Workflow();
    
    // 注册Agent
    this.agentSystem.registerAgent(new CodeAnalysisAgent());
    this.agentSystem.registerAgent(new TestGenerationAgent());
    this.agentSystem.registerAgent(new CodeOptimizationAgent());
  }
  
  async assistWithProblem(problem, userCode) {
    const workflow = this.createWorkflow(problem, userCode);
    
    const results = {};
    
    // 执行工作流
    for (const step of workflow.steps) {
      const agent = this.agentSystem.agents.get(step.agentId);
      const result = await agent.execute(step.task);
      results[step.name] = result;
      
      // 根据结果决定下一步
      if (step.nextStepCondition && !step.nextStepCondition(result)) {
        break;
      }
    }
    
    return this.generateFinalReport(results);
  }
  
  createWorkflow(problem, userCode) {
    return {
      steps: [
        {
          name: 'syntax-check',
          agentId: 'code-analysis',
          task: { code: userCode, language: 'javascript' },
          nextStepCondition: (result) => result.syntaxErrors.length === 0
        },
        {
          name: 'test-generation',
          agentId: 'test-generation',
          task: { 
            code: userCode, 
            functionSignature: problem.functionSignature,
            examples: problem.examples 
          }
        },
        {
          name: 'optimization',
          agentId: 'code-optimization',
          task: { 
            code: userCode, 
            language: 'javascript',
            optimizationGoal: 'performance' 
          }
        }
      ]
    };
  }
}
```

```javascript
class TestCaseDesigner {
  constructor(problem) {
    this.problem = problem;
    this.testCases = [];
  }
  
  // 1. 基于示例生成测试用例
  generateFromExamples() {
    return this.problem.examples.map((example, index) => ({
      id: `example-${index}`,
      input: example.input,
      expectedOutput: example.output,
      isExample: true,
      isHidden: false,
      description: `示例 ${index + 1}`
    }));
  }
  
  // 2. 边界值测试
  generateBoundaryTests() {
    const tests = [];
    const constraints = this.parseConstraints();
    
    // 数值边界
    if (constraints.numeric) {
      tests.push({
        id: 'boundary-min',
        input: this.generateMinInput(constraints.numeric),
        expectedOutput: null, // 需要计算
        description: '最小值边界'
      });
      
      tests.push({
        id: 'boundary-max',
        input: this.generateMaxInput(constraints.numeric),
        expectedOutput: null,
        description: '最大值边界'
      });
    }
    
    // 数组边界
    if (constraints.array) {
      tests.push({
        id: 'boundary-empty-array',
        input: '[]',
        expectedOutput: null,
        description: '空数组'
      });
      
      tests.push({
        id: 'boundary-single-element',
        input: '[1]',
        expectedOutput: null,
        description: '单元素数组'
      });
    }
    
    // 字符串边界
    if (constraints.string) {
      tests.push({
        id: 'boundary-empty-string',
        input: '""',
        expectedOutput: null,
        description: '空字符串'
      });
    }
    
    return tests;
  }
  
  // 3. 特殊情况测试
  generateSpecialCases() {
    const tests = [];
    
    // 常见特殊情况
    const specialCases = [
      { name: 'all-zeros', generator: () => this.generateAllZeros() },
      { name: 'all-same', generator: () => this.generateAllSame() },
      { name: 'sorted-asc', generator: () => this.generateSortedAsc() },
      { name: 'sorted-desc', generator: () => this.generateSortedDesc() },
      { name: 'reverse-sorted', generator: () => this.generateReverseSorted() }
    ];
    
    specialCases.forEach(specialCase => {
      tests.push({
        id: `special-${specialCase.name}`,
        input: specialCase.generator(),
        expectedOutput: null,
        description: `特殊情况: ${specialCase.name}`
      });
    });
    
    return tests;
  }
  
  // 4. 随机测试
  generateRandomTests(count = 10) {
    const tests = [];
    
    for (let i = 0; i < count; i++) {
      tests.push({
        id: `random-${i}`,
        input: this.generateRandomInput(),
        expectedOutput: null,
        description: `随机测试 ${i + 1}`
      });
    }
    
    return tests;
  }
  
  // 5. 性能测试
  generatePerformanceTests() {
    const tests = [];
    const sizes = [100, 1000, 10000, 100000];
    
    sizes.forEach(size => {
      tests.push({
        id: `perf-${size}`,
        input: this.generateLargeInput(size),
        expectedOutput: null,
        description: `性能测试: ${size} 规模`,
        timeLimit: this.calculateTimeLimit(size),
        memoryLimit: this.calculateMemoryLimit(size)
      });
    });
    
    return tests;
  }
  
  // 6. 错误输入测试
  generateInvalidInputTests() {
    const tests = [];
    
    // 无效输入
    const invalidInputs = [
      { input: 'null', description: 'null输入' },
      { input: 'undefined', description: 'undefined输入' },
      { input: '{}', description: '空对象' },
      { input: '[]', description: '空数组（如果期望非空）' }
    ];
    
    invalidInputs.forEach(invalid => {
      tests.push({
        id: `invalid-${invalid.description}`,
        input: invalid.input,
        expectedOutput: null,

  }
  
  // 综合测试用例生成
  generateAllTestCases() {

      ...this.generateBoundaryTests(),

  }
}

class TestCaseExecutor {
  constructor(solution, language) {
    this.solution = solution;
    this.language = language;
    this.sandbox = new Sandbox();
  }
  
  async execute(testCase) {
    const startTime = Date.now();
    const startMemory = process.memoryUsage().heapUsed;
    
    try {
      // 在沙箱中执行
      const result = await this.sandbox.execute(
        this.solution,
        testCase.input,
        this.language,
        {
          timeout: testCase.timeLimit || 5000,
          memoryLimit: testCase.memoryLimit || 256 * 1024 * 1024
        }
      );
      
      const endTime = Date.now();
      const endMemory = process.memoryUsage().heapUsed;
      
      return {
        testCaseId: testCase.id,
        passed: this.compareOutputs(result.output, testCase.expectedOutput),
        actualOutput: result.output,
        executionTime: endTime - startTime,
        memoryUsage: endMemory - startMemory,
        error: result.error
      };
      
    } catch (error) {
      return {
        testCaseId: testCase.id,
        passed: false,
        actualOutput: null,
        executionTime: Date.now() - startTime,
        memoryUsage: 0,
        error: error.message
      };
    }
  }
  
  compareOutputs(actual, expected) {
    // 标准化输出
    const normalize = (str) => str.trim().replace(/\s+/g, ' ');
    return normalize(actual) === normalize(expected);
  }
}
```

**解题状态机设计：**

```javascript
class ProblemSolvingStateMachine {
  constructor() {
    this.states = {
      IDLE: 'idle',
      READING: 'reading',
      ANALYZING: 'analyzing',
      CODING: 'coding',
      TESTING: 'testing',
      DEBUGGING: 'debugging',
      SUBMITTING: 'submitting',
      REVIEWING: 'reviewing',
      COMPLETED: 'completed',
      FAILED: 'failed'
    };
    
    this.currentState = this.states.IDLE;
    this.context = {};
    this.history = [];
    this.transitions = this.defineTransitions();
  }
  
  defineTransitions() {
    return {
      [this.states.IDLE]: {
        START_READING: {
          target: this.states.READING,
          action: () => this.startReading()
        }
      },
      [this.states.READING]: {
        FINISH_READING: {
          target: this.states.ANALYZING,
          action: () => this.finishReading()
        },
        ABANDON: {
          target: this.states.IDLE,
          action: () => this.abandon()
        }
      },
      [this.states.ANALYZING]: {
        START_CODING: {
          target: this.states.CODING,
          action: () => this.startCoding()
        },
        REQUEST_HINT: {
          target: this.states.ANALYZING, // 保持在分析状态
          action: () => this.requestHint()
        }
      },
      [this.states.CODING]: {
        SUBMIT_CODE: {
          target: this.states.TESTING,
          action: () => this.submitCode()
        },
        SAVE_DRAFT: {
          target: this.states.CODING, // 保持在编码状态
          action: () => this.saveDraft()
        },
        REQUEST_HELP: {
          target: this.states.CODING,
          action: () => this.requestHelp()
        }
      },
      [this.states.TESTING]: {
        ALL_TESTS_PASSED: {
          target: this.states.SUBMITTING,
          action: () => this.allTestsPassed()
        },
        SOME_TESTS_FAILED: {
          target: this.states.DEBUGGING,
          action: () => this.someTestsFailed()
        },
        RUNTIME_ERROR: {
          target: this.states.DEBUGGING,
          action: () => this.runtimeError()
        }
      },
      [this.states.DEBUGGING]: {
        FIX_CODE: {
          target: this.states.CODING,
          action: () => this.fixCode()
        },
        REQUEST_DEBUG_HELP: {
          target: this.states.DEBUGGING,
          action: () => this.requestDebugHelp()
        },
        GIVE_UP: {
          target: this.states.FAILED,
          action: () => this.giveUp()
        }
      },
      [this.states.SUBMITTING]: {
        SUBMISSION_SUCCESS: {
          target: this.states.REVIEWING,
          action: () => this.submissionSuccess()
        },
        SUBMISSION_FAILED: {
          target: this.states.DEBUGGING,
          action: () => this.submissionFailed()
        }
      },
      [this.states.REVIEWING]: {
        ACCEPT_SOLUTION: {
          target: this.states.COMPLETED,
          action: () => this.acceptSolution()
        },
        REJECT_SOLUTION: {
          target: this.states.CODING,
          action: () => this.rejectSolution()
        }
      }
    };
  }
  
  // 触发事件
  async trigger(event, data = {}) {
    const transition = this.transitions[this.currentState]?.[event];
    
    if (!transition) {
      throw new Error(`无效的状态转换: ${this.currentState} -> ${event}`);
    }
    
    // 记录历史
    this.history.push({
      from: this.currentState,
      to: transition.target,
      event,
      data,
      timestamp: Date.now()
    });
    
    // 执行转换动作
    if (transition.action) {
      await transition.action(data);
    }
    
    // 更新状态
    this.currentState = transition.target;
    this.context = { ...this.context, ...data };
    
    return this.currentState;
```

```javascript
class AgentSecurityFramework {
  constructor() {
    this.securityLayers = [
      new InputValidationLayer(),
      new PermissionControlLayer(),
      new AuditLoggingLayer(),
      new RateLimitingLayer(),
      new ContentFilterLayer()
    ];
  }
  
  // 多层安全检查
  async checkSecurity(context) {
    const results = [];
    
    for (const layer of this.securityLayers) {
      const result = await layer.check(context);
      results.push(result);
      
      if (!result.passed) {
        return {
          passed: false,
          failedLayer: layer.name,
          reason: result.reason
        };
      }
    }
    
    return { passed: true, results };
  }

// 1. 输入验证层
class InputValidationLayer {
  constructor() {
    this.name = 'input-validation';
    this.patterns = {
      sqlInjection: /(\b(SELECT|INSERT|UPDATE|DELETE|DROP|CREATE|ALTER)\b)/i,
      xss: /<script\b[^>]*>([\s\S]*?)<\/script>/gi,
      commandInjection: /[;&|`$]/,
      pathTraversal: /\.\.\//
    };
  }
  
  async check(context) {
    const { input } = context;
    
    // SQL注入检测
    if (this.patterns.sqlInjection.test(input)) {
      return { passed: false, reason: '检测到SQL注入尝试' };
    }
    
    // XSS检测
    if (this.patterns.xss.test(input)) {
      return { passed: false, reason: '检测到XSS攻击尝试' };
    }
    
    // 命令注入检测
    if (this.patterns.commandInjection.test(input)) {
      return { passed: false, reason: '检测到命令注入尝试' };
    }
    
    // 路径遍历检测
    if (this.patterns.pathTraversal.test(input)) {
      return { passed: false, reason: '检测到路径遍历尝试' };
    }
    
    // 长度检查
    if (input.length > 10000) {

class RateLimitingLayer {
  constructor() {
    this.name = 'rate-limiting';
    this.limits = new Map();
    this.requests = new Map();
  }
  
  // 设置限流规则
  setLimit(key, { maxRequests, windowMs }) {
    this.limits.set(key, { maxRequests, windowMs });
  }
  
  async check(context) {
    const { userId, action } = context;
    const key = `${userId}:${action}`;
    const limit = this.limits.get(key) || { maxRequests: 100, windowMs: 60000 };
    
    const now = Date.now();
    const windowStart = now - limit.windowMs;
    
    // 获取时间窗口内的请求
    if (!this.requests.has(key)) {
      this.requests.set(key, []);
    }
    
    const requests = this.requests.get(key);
    const recentRequests = requests.filter(time => time > windowStart);
    
    if (recentRequests.length >= limit.maxRequests) {
      return { 
        passed: false, 
        reason: `请求过于频繁，限制 ${limit.maxRequests} 次/分钟` 
      };
    }
    
    // 记录请求
    recentRequests.push(now);
    this.requests.set(key, recentRequests);
    
    return { passed: true };
  }
}

class ContentFilterLayer {
  constructor() {
    this.name = 'content-filter';
    this.filters = [
      this.filterSensitiveInfo.bind(this),
      this.filterHarmfulContent.bind(this),
      this.filterPromptInjection.bind(this)
    ];
  }
  
  async check(context) {
    const { input } = context;
    
    for (const filter of this.filters) {
      const result = await filter(input);
      if (!result.passed) {
        return result;
      }
    }
    
    return { passed: true };
  }
  
  async filterSensitiveInfo(input) {
    const patterns = [
      { type: '身份证号', regex: /\b\d{18}\b/ },
      { type: '手机号', regex: /\b1[3-9]\d{9}\b/ },
      { type: '银行卡号', regex: /\b\d{16,19}\b/ },
      { type: '邮箱', regex: /\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b/ }
    ];
    
    for (const pattern of patterns) {
      if (pattern.regex.test(input)) {
        return { 
          passed: false, 
          reason: `检测到敏感信息: ${pattern.type}` 
        };
      }
    }
    
    return { passed: true };
  }
  
  async filterHarmfulContent(input) {
    const harmfulKeywords = ['暴力', '色情', '赌博', '毒品', '诈骗'];
    
    for (const keyword of harmfulKeywords) {
      if (input.includes(keyword)) {
        return { 
          passed: false, 
          reason: `检测到有害内容: ${keyword}` 
        };
      }
    }
    
    return { passed: true };
  }
  
  async filterPromptInjection(input) {
    const injectionPatterns = [
      /ignore previous instructions/i,
      /forget your instructions/i,
      /you are now/i,
      /system prompt/i,
      /reveal your instructions/i
    ];
    
    for (const pattern of injectionPatterns) {
      if (pattern.test(input)) {
        return { 
          passed: false, 
          reason: '检测到提示词注入尝试' 
        };
      }
    }
    
    return { passed: true };
  }
}
```
